//! Archive extraction (zip and tar.gz) that treats every entry name as hostile.
//!
//! Each entry's destination is built with [`SafePath`], which rejects `..`, absolute paths,
//! Windows device names and anything that resolves outside the target folder through a
//! symlink. Hard links are refused; symlinks are only created when their target stays
//! inside the folder. A total-size cap stops decompression bombs.

use std::collections::HashSet;
use std::fs::File;
use std::io::{self, Read};
use std::path::Path;

use crate::paths::{validate_relative, SafePath};
use crate::{Error, Result};

/// Default cap on the total unpacked size: more than any JDK or natives bundle we handle.
pub const DEFAULT_MAX_UNPACKED: u64 = 2 * 1024 * 1024 * 1024;

pub struct ExtractOptions<'a> {
    /// Leading path components to drop from every entry (`1` removes a JDK's top folder).
    pub strip_components: usize,
    pub max_total_bytes: u64,
    /// Return `false` to skip an entry. Receives the entry's original name.
    pub filter: Option<&'a dyn Fn(&str) -> bool>,
}

impl Default for ExtractOptions<'_> {
    fn default() -> Self {
        Self {
            strip_components: 0,
            max_total_bytes: DEFAULT_MAX_UNPACKED,
            filter: None,
        }
    }
}

fn bad(msg: impl Into<String>) -> Error {
    Error::Archive(msg.into())
}

impl From<zip::result::ZipError> for Error {
    fn from(e: zip::result::ZipError) -> Self {
        Self::Archive(e.to_string())
    }
}

/// Where an entry goes, or `None` if it is the stripped-away top folder itself.
fn destination(root: &Path, name: &str, strip: usize) -> Result<Option<SafePath>> {
    // Validate the whole original name first so a `..` hidden in a stripped component
    // is still caught.
    let parts = validate_relative(name)?;
    if parts.len() <= strip {
        return Ok(None);
    }
    Ok(Some(SafePath::new(root, &parts[strip..].join("/"))?))
}

#[cfg(unix)]
fn set_mode(path: &Path, executable: bool) -> Result<()> {
    use std::os::unix::fs::PermissionsExt;
    let mode = if executable { 0o755 } else { 0o644 };
    std::fs::set_permissions(path, std::fs::Permissions::from_mode(mode))?;
    Ok(())
}

#[cfg(not(unix))]
fn set_mode(_path: &Path, _executable: bool) -> Result<()> {
    Ok(())
}

/// Copy at most the remaining budget from `reader`, failing if the entry is larger.
fn copy_limited(reader: &mut impl Read, dest: &Path, total: &mut u64, max: u64) -> Result<()> {
    let mut out = File::create(dest)?;
    let remaining = max.saturating_sub(*total);
    let copied = io::copy(&mut reader.take(remaining.saturating_add(1)), &mut out)?;
    *total += copied;
    if *total > max {
        drop(out);
        let _ = std::fs::remove_file(dest);
        return Err(bad("the archive is larger than allowed"));
    }
    Ok(())
}

/// Read one named entry of a zip fully into memory, refusing entries over `max_bytes`.
pub fn read_zip_entry(archive: &Path, name: &str, max_bytes: u64) -> Result<Vec<u8>> {
    let mut zip = zip::ZipArchive::new(File::open(archive)?)?;
    let entry = zip.by_name(name)?;
    let mut out = Vec::new();
    entry
        .take(max_bytes.saturating_add(1))
        .read_to_end(&mut out)?;
    if out.len() as u64 > max_bytes {
        return Err(bad("an entry in the archive is larger than allowed"));
    }
    Ok(out)
}

/// Extract a zip into `root` (created if needed). Returns the number of files written.
pub fn extract_zip(archive: &Path, root: &Path, opts: &ExtractOptions<'_>) -> Result<usize> {
    std::fs::create_dir_all(root)?;
    let mut zip = zip::ZipArchive::new(File::open(archive)?)?;
    let (mut files, mut total) = (0usize, 0u64);
    for i in 0..zip.len() {
        let mut entry = zip.by_index(i)?;
        let name = entry.name().to_owned();
        if opts.filter.is_some_and(|f| !f(&name)) {
            continue;
        }
        let mode = entry.unix_mode();
        if mode.is_some_and(|m| m & 0o170_000 == 0o120_000) {
            return Err(bad("the archive contains a symbolic link"));
        }
        let Some(dest) = destination(root, &name, opts.strip_components)? else {
            continue;
        };
        if entry.is_dir() {
            std::fs::create_dir_all(dest.as_path())?;
            continue;
        }
        if let Some(parent) = dest.as_path().parent() {
            std::fs::create_dir_all(parent)?;
        }
        copy_limited(&mut entry, dest.as_path(), &mut total, opts.max_total_bytes)?;
        set_mode(dest.as_path(), mode.is_some_and(|m| m & 0o111 != 0))?;
        files += 1;
    }
    Ok(files)
}

/// Zip up `src_dir` into `dest`, naming entries `prefix` + relative path. Top-level names in
/// `skip_top` are left out, symlinks are never followed, and `first` entries (name, bytes)
/// are written before the folder contents. The archive appears at `dest` only when complete.
pub fn write_zip(
    dest: &Path,
    prefix: &str,
    src_dir: &Path,
    skip_top: &[&str],
    first: &[(String, Vec<u8>)],
) -> Result<()> {
    write_zip_excluding(dest, prefix, src_dir, skip_top, first, &HashSet::new())
}

/// Like [`write_zip`], and also leaves out files whose path relative to `src_dir` (with `/`
/// separators) is in `exclude`.
pub fn write_zip_excluding(
    dest: &Path,
    prefix: &str,
    src_dir: &Path,
    skip_top: &[&str],
    first: &[(String, Vec<u8>)],
    exclude: &HashSet<String>,
) -> Result<()> {
    use zip::write::SimpleFileOptions;
    fn walk(
        w: &mut zip::ZipWriter<File>,
        dir: &Path,
        rel: &str,
        top: bool,
        prefix: &str,
        skip: &[&str],
        exclude: &HashSet<String>,
    ) -> Result<()> {
        let mut entries: Vec<_> = std::fs::read_dir(dir)?.collect::<std::io::Result<_>>()?;
        entries.sort_by_key(std::fs::DirEntry::file_name);
        for e in entries {
            let name = e.file_name().to_string_lossy().into_owned();
            if top && skip.contains(&name.as_str()) {
                continue;
            }
            let rel_name = if rel.is_empty() {
                name
            } else {
                format!("{rel}/{name}")
            };
            let kind = e.file_type()?;
            if kind.is_dir() {
                walk(w, &e.path(), &rel_name, false, prefix, skip, exclude)?;
            } else if kind.is_file() && !exclude.contains(&rel_name) {
                let big = e.metadata()?.len() > 3 * 1024 * 1024 * 1024;
                let opts = SimpleFileOptions::default()
                    .compression_method(zip::CompressionMethod::Deflated)
                    .large_file(big);
                w.start_file(format!("{prefix}{rel_name}"), opts)?;
                io::copy(&mut File::open(e.path())?, w)?;
            }
        }
        Ok(())
    }
    let tmp = dest.with_extension("partial");
    let result = (|| -> Result<()> {
        let mut w = zip::ZipWriter::new(File::create(&tmp)?);
        let opts =
            SimpleFileOptions::default().compression_method(zip::CompressionMethod::Deflated);
        for (name, data) in first {
            w.start_file(name.as_str(), opts)?;
            io::Write::write_all(&mut w, data)?;
        }
        walk(&mut w, src_dir, "", true, prefix, skip_top, exclude)?;
        w.finish()?;
        Ok(())
    })();
    match result {
        Ok(()) => Ok(std::fs::rename(&tmp, dest)?),
        Err(e) => {
            let _ = std::fs::remove_file(&tmp);
            Err(e)
        }
    }
}

/// Lexically check that a symlink at `link_rel` pointing at `target` stays inside the root.
fn symlink_stays_inside(link_rel: &[&str], target: &str) -> bool {
    if target.is_empty()
        || target.starts_with('/')
        || target.starts_with('\\')
        || target.contains(':')
    {
        return false;
    }
    let mut stack: Vec<&str> = link_rel[..link_rel.len().saturating_sub(1)].to_vec();
    for part in target.split(['/', '\\']) {
        match part {
            "" | "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return false;
                }
            }
            p => stack.push(p),
        }
    }
    true
}

/// Extract a `.tar.gz` into `root` (created if needed). Returns the number of files written.
pub fn extract_tar_gz(archive: &Path, root: &Path, opts: &ExtractOptions<'_>) -> Result<usize> {
    std::fs::create_dir_all(root)?;
    let gz = flate2::read::GzDecoder::new(File::open(archive)?);
    let mut tar = tar::Archive::new(gz);
    let (mut files, mut total) = (0usize, 0u64);
    for entry in tar.entries()? {
        let mut entry = entry?;
        let name = entry.path()?.to_string_lossy().into_owned();
        let kind = entry.header().entry_type();
        if kind.is_hard_link() {
            return Err(bad("the archive contains a hard link"));
        }
        if !(kind.is_file() || kind.is_dir() || kind.is_symlink()) {
            continue; // pax headers, devices and the like carry no files we want
        }
        if opts.filter.is_some_and(|f| !f(&name)) {
            continue;
        }
        let Some(dest) = destination(root, &name, opts.strip_components)? else {
            continue;
        };
        if kind.is_dir() {
            std::fs::create_dir_all(dest.as_path())?;
            continue;
        }
        if let Some(parent) = dest.as_path().parent() {
            std::fs::create_dir_all(parent)?;
        }
        if kind.is_symlink() {
            let target = entry
                .link_name()?
                .map(|t| t.to_string_lossy().into_owned())
                .unwrap_or_default();
            let rel = dest.relative().to_string_lossy().replace('\\', "/");
            let rel_parts: Vec<&str> = rel.split('/').collect();
            if !symlink_stays_inside(&rel_parts, &target) {
                return Err(bad(
                    "the archive contains a link that points outside its folder",
                ));
            }
            #[cfg(unix)]
            {
                let _ = std::fs::remove_file(dest.as_path());
                std::os::unix::fs::symlink(&target, dest.as_path())?;
            }
            continue;
        }
        let executable = entry.header().mode().is_ok_and(|m| m & 0o111 != 0);
        copy_limited(&mut entry, dest.as_path(), &mut total, opts.max_total_bytes)?;
        set_mode(dest.as_path(), executable)?;
        files += 1;
    }
    Ok(files)
}

mod tests {
    use std::io::Write;

    use super::*;

    fn make_zip(path: &Path, entries: &[(&str, &[u8])]) {
        let mut w = zip::ZipWriter::new(File::create(path).unwrap());
        let opts = zip::write::SimpleFileOptions::default();
        for (name, data) in entries {
            w.start_file(*name, opts).unwrap();
            w.write_all(data).unwrap();
        }
        w.finish().unwrap();
    }

    fn make_tar_gz(
        path: &Path,
        build: impl FnOnce(&mut tar::Builder<flate2::write::GzEncoder<File>>),
    ) {
        let gz =
            flate2::write::GzEncoder::new(File::create(path).unwrap(), flate2::Compression::fast());
        let mut b = tar::Builder::new(gz);
        build(&mut b);
        b.into_inner().unwrap().finish().unwrap();
    }

    /// Append an entry with a raw, unvalidated name (the builder refuses `..` itself).
    fn append_raw(b: &mut tar::Builder<flate2::write::GzEncoder<File>>, name: &str, data: &[u8]) {
        let mut h = tar::Header::new_gnu();
        h.as_gnu_mut().unwrap().name[..name.len()].copy_from_slice(name.as_bytes());
        h.set_size(data.len() as u64);
        h.set_mode(0o644);
        h.set_entry_type(tar::EntryType::Regular);
        h.set_cksum();
        b.append(&h, data).unwrap();
    }

    fn dirs() -> (tempfile::TempDir, std::path::PathBuf, std::path::PathBuf) {
        let base = tempfile::tempdir().unwrap();
        let out = base.path().join("out");
        let zip = base.path().join("a.archive");
        (base, out, zip)
    }

//! Parallel, resumable, verified downloads.
//!
//! Every file is written to `<dest>.part`, hashed while it is written, checked against its
//! expected digest, and only then renamed over `<dest>`, so a file at its final path has
//! always passed verification. Interrupted downloads resume with a `Range` request.
//! Destinations are [`SafePath`]s: callers can't hand this module a raw path.

use std::collections::HashSet;
use std::path::{Path, PathBuf};
use std::sync::atomic::{AtomicU64, AtomicUsize, Ordering};
use std::sync::Arc;
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use serde::Serialize;
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::sync::{watch, Semaphore};
use tokio::task::JoinSet;

use crate::hash::{ExpectedHash, StreamHasher};
use crate::net::HttpClient;
use crate::paths::SafePath;
use crate::{Error, Result};

/// One file to fetch.
#[derive(Debug, Clone)]
pub struct DownloadTask {
    pub url: String,
    pub dest: SafePath,
    /// Verified before the file is kept. Strongly preferred.
    pub hash: Option<ExpectedHash>,
    /// Expected size in bytes. Used for progress, for cheap "already have it" checks and to
    /// reject over-long responses.
    pub size: Option<u64>,
}

#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize)]
pub struct Progress {
    pub bytes_done: u64,
    pub bytes_total: u64,
    pub files_done: usize,
    pub files_total: usize,
}

#[derive(Debug, Clone)]
pub struct RetryPolicy {
    pub attempts: u32,
    pub base_delay: Duration,
}

impl Default for RetryPolicy {
    fn default() -> Self {
        Self {
            attempts: 4,
            base_delay: Duration::from_millis(500),
        }
    }
}

impl RetryPolicy {
    /// Exponential backoff with up to 50% jitter, capped at 30 s.
    fn delay(&self, attempt: u32) -> Duration {
        let exp = self
            .base_delay
            .saturating_mul(2u32.saturating_pow(attempt))
            .min(Duration::from_secs(30));
        let nanos = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .map_or(0, |d| u64::from(d.subsec_nanos()));
        // Cheap jitter: no need for a random-number crate to desynchronize retries.
        let jitter = exp.as_nanos() as u64 / 2;
        let extra = if jitter == 0 { 0 } else { nanos % jitter };
        exp + Duration::from_nanos(extra)
    }
}

/// Why one attempt failed, and whether trying again makes sense.
enum Failure {
    Retry(Error),
    Fatal(Error),
}

struct Tracker {
    bytes_done: AtomicU64,
    bytes_total: u64,
    files_done: AtomicUsize,
    files_total: usize,
    tx: Option<watch::Sender<Progress>>,
}

impl Tracker {
    fn publish(&self) {
        if let Some(tx) = &self.tx {
            tx.send_replace(Progress {
                bytes_done: self.bytes_done.load(Ordering::Relaxed),
                bytes_total: self.bytes_total,
                files_done: self.files_done.load(Ordering::Relaxed),
                files_total: self.files_total,
            });
        }
    }

    fn add(&self, n: u64) {
        self.bytes_done.fetch_add(n, Ordering::Relaxed);
        self.publish();
    }

    fn sub(&self, n: u64) {
        let _ = self
            .bytes_done
            .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |v| {
                Some(v.saturating_sub(n))
            });
        self.publish();
    }

    fn file_done(&self) {
        self.files_done.fetch_add(1, Ordering::Relaxed);
        self.publish();
    }
}

#[derive(Debug, Clone)]
pub struct Downloader {
    http: HttpClient,
    max_concurrent: usize,
    retry: RetryPolicy,
}

fn part_path(dest: &SafePath) -> PathBuf {
    let mut name = dest.as_path().as_os_str().to_owned();
    name.push(".part");
    PathBuf::from(name)
}

fn io_err(e: std::io::Error) -> Error {
    Error::Io(e)
}

impl Downloader {
    /// `max_concurrent` is the `max_concurrent_downloads` setting (clamped to 1..=16).
    pub fn new(http: HttpClient, max_concurrent: u8) -> Self {
        Self {
            http,
            max_concurrent: usize::from(max_concurrent).clamp(1, 16),
            retry: RetryPolicy::default(),
        }
    }

    pub fn with_retry(mut self, retry: RetryPolicy) -> Self {
        self.retry = retry;
        self
    }

    /// Download every task, at most `max_concurrent` at a time. Stops at the first failure.
    /// Tasks with the same destination are fetched once.
    pub async fn download_all(
        &self,
        tasks: Vec<DownloadTask>,
        progress: Option<watch::Sender<Progress>>,
    ) -> Result<()> {
        let mut seen = HashSet::new();
        let tasks: Vec<DownloadTask> = tasks
            .into_iter()
            .filter(|t| seen.insert(t.dest.as_path().to_path_buf()))
            .collect();
        let tracker = Arc::new(Tracker {
            bytes_done: AtomicU64::new(0),
            bytes_total: tasks.iter().filter_map(|t| t.size).sum(),
            files_done: AtomicUsize::new(0),
            files_total: tasks.len(),
            tx: progress,
        });
        tracker.publish();

        let this = Arc::new(self.clone());
        let limit = Arc::new(Semaphore::new(self.max_concurrent));
        let mut set = JoinSet::new();
        for task in tasks {
            let (this, limit, tracker) = (this.clone(), limit.clone(), tracker.clone());
            set.spawn(async move {
                let _permit = limit
                    .acquire_owned()
                    .await
                    .map_err(|e| Error::Io(std::io::Error::other(e)))?;
                this.download_one(&task, &tracker).await
            });
        }
        while let Some(joined) = set.join_next().await {
            let outcome = joined.map_err(|e| Error::Io(std::io::Error::other(e)));
            if let Err(e) | Ok(Err(e)) = outcome {
                set.abort_all();
                return Err(e);
            }
        }
        Ok(())
    }

    /// Download a single file with no progress reporting.
    pub async fn download(&self, task: DownloadTask) -> Result<()> {
        self.download_all(vec![task], None).await
    }

    async fn download_one(&self, task: &DownloadTask, tracker: &Tracker) -> Result<()> {
        if let Some(parent) = task.dest.as_path().parent() {
            tokio::fs::create_dir_all(parent).await.map_err(io_err)?;
        }
        if is_already_present(task).await {
            if let Some(size) = task.size {
                tracker.add(size);
            }
            tracker.file_done();
            return Ok(());
        }
        let mut counted = 0u64;
        for attempt in 0..self.retry.attempts.max(1) {
            match self.attempt(task, tracker, &mut counted).await {
                Ok(()) => {
                    tracker.file_done();
                    return Ok(());
                }
                Err(Failure::Fatal(e)) => return Err(e),
                Err(Failure::Retry(e)) => {
                    if attempt + 1 >= self.retry.attempts.max(1) {
                        return Err(e);
                    }
                    tracing::debug!(attempt, "download attempt failed; retrying");
                    tokio::time::sleep(self.retry.delay(attempt)).await;
                }
            }
        }
        Err(Error::Http("download failed".into()))
    }

    async fn attempt(
        &self,
        task: &DownloadTask,
        tracker: &Tracker,
        counted: &mut u64,
    ) -> std::result::Result<(), Failure> {
        let part = part_path(&task.dest);
        let mut existing = match tokio::fs::metadata(&part).await {
            Ok(m) if m.is_file() => m.len(),
            _ => 0,
        };
        // A leftover longer than the file can't be a prefix of it.
        if task.size.is_some_and(|s| existing > s) {
            discard(&part, tracker, counted).await;
            existing = 0;
        }
        // Bytes counted by a failed earlier attempt are re-counted from the file.
        tracker.sub(*counted);
        *counted = 0;

        let mut hasher = task.hash.as_ref().map(|h| StreamHasher::new(h.algo()));
        let mut written = 0u64;

        let complete_already = task.size == Some(existing) && existing > 0;
        if complete_already {
            written = prefix_into(&part, &mut hasher, tracker, counted).await?;
        } else {
            let mut stream = self
                .http
                .get_stream(&task.url, (existing > 0).then_some(existing))
                .await
                .map_err(Failure::Retry)?;
            let resuming = match stream.status {
                206 if stream.content_range_start == Some(existing) && existing > 0 => true,
                200 => false,
                416 => {
                    discard(&part, tracker, counted).await;
                    return Err(Failure::Retry(Error::Http(
                        "the server rejected a resumed download".into(),
                    )));
                }
                206 => {
                    discard(&part, tracker, counted).await;
                    return Err(Failure::Retry(Error::Http(
                        "the server resumed at the wrong position".into(),
                    )));
                }
                429 | 500..=599 => {
                    return Err(Failure::Retry(Error::Http(format!(
                        "the server is busy (HTTP {})",
                        stream.status
                    ))));
                }
                status => {
                    return Err(Failure::Fatal(Error::Http(format!(
                        "the server refused the download (HTTP {status})"
                    ))));
                }
            };

            let mut file = if resuming {
                written = prefix_into(&part, &mut hasher, tracker, counted).await?;
                tokio::fs::OpenOptions::new()
                    .append(true)
                    .open(&part)
                    .await
                    .map_err(|e| Failure::Retry(io_err(e)))?
            } else {
                tokio::fs::File::create(&part)
                    .await
                    .map_err(|e| Failure::Retry(io_err(e)))?
            };

            while let Some(chunk) = stream.next_chunk().await.map_err(Failure::Retry)? {
                written += chunk.len() as u64;
                if task.size.is_some_and(|s| written > s) {
                    drop(file);
                    discard(&part, tracker, counted).await;
                    return Err(Failure::Retry(Error::Http(
                        "the server sent more data than expected".into(),
                    )));
                }
                if let Some(h) = hasher.as_mut() {
                    h.update(&chunk);
                }
                file.write_all(&chunk)
                    .await
                    .map_err(|e| Failure::Fatal(io_err(e)))?;
                tracker.add(chunk.len() as u64);
                *counted += chunk.len() as u64;
            }
            file.flush().await.map_err(|e| Failure::Fatal(io_err(e)))?;
        }

        // A short body keeps its .part so the next attempt can resume.
        if task.size.is_some_and(|s| written != s) {
            return Err(Failure::Retry(Error::Http(
                "the download ended early".into(),
            )));
        }
        if let (Some(hasher), Some(expected)) = (hasher, &task.hash) {
            let actual = hasher.finalize_hex();
            if actual != expected.hex() {
                discard(&part, tracker, counted).await;
                return Err(Failure::Retry(Error::HashMismatch {
                    expected: expected.hex().to_owned(),
                    actual,
                }));
            }
        }
        tokio::fs::rename(&part, task.dest.as_path())
            .await
            .map_err(|e| Failure::Fatal(io_err(e)))?;
        Ok(())
    }
}

async fn is_already_present(task: &DownloadTask) -> bool {
    let Ok(meta) = tokio::fs::metadata(task.dest.as_path()).await else {
        return false;
    };
    if !meta.is_file() {
        return false;
    }
    // Files only reach their final path after verification, so a size match is enough
    // to skip them cheaply (this runs for thousands of assets on every launch).
    if let Some(size) = task.size {
        return meta.len() == size;
    }
    match &task.hash {
        Some(h) => crate::hash::verify_file(task.dest.as_path(), h)
            .await
            .is_ok(),
        None => false,
    }
}

async fn discard(part: &Path, tracker: &Tracker, counted: &mut u64) {
    let _ = tokio::fs::remove_file(part).await;
    tracker.sub(*counted);
    *counted = 0;
}

/// Feed the bytes already on disk into the hasher and count them as done.
async fn prefix_into(
    part: &Path,
    hasher: &mut Option<StreamHasher>,
    tracker: &Tracker,
    counted: &mut u64,
) -> std::result::Result<u64, Failure> {
    let mut file = tokio::fs::File::open(part)
        .await
        .map_err(|e| Failure::Retry(io_err(e)))?;
    let mut buf = vec![0u8; 64 * 1024];
    let mut total = 0u64;
    loop {
        let n = file
            .read(&mut buf)
            .await
            .map_err(|e| Failure::Retry(io_err(e)))?;
        if n == 0 {
            break;
        }
        if let Some(h) = hasher.as_mut() {
            h.update(&buf[..n]);
        }
        total += n as u64;
    }
    tracker.add(total);
    *counted += total;
    Ok(total)
}

/// Put `src` at `dst` as a hard link, falling back to a copy when the two are on different
/// filesystems (or the filesystem has no hard links). Used to share cached files between
/// instances without duplicating them.
pub fn link_or_copy(src: &Path, dst: &SafePath) -> Result<()> {
    if let Some(parent) = dst.as_path().parent() {
        std::fs::create_dir_all(parent)?;
    }
    match std::fs::remove_file(dst.as_path()) {
        Ok(()) => {}
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {}
        Err(e) => return Err(e.into()),
    }
    if std::fs::hard_link(src, dst.as_path()).is_err() {
        std::fs::copy(src, dst.as_path())?;
    }
    Ok(())
}

/// Extract the major version from the output of `java -version`.
pub fn parse_java_major(output: &str) -> Option<u32> {
    let start = output.find("version \"")? + "version \"".len();
    let rest = &output[start..];
    let version = &rest[..rest.find('"')?];
    let mut parts = version.split(['.', '-', '_', '+']);
    let first: u32 = parts.next()?.parse().ok()?;
    // Java 8 and earlier call themselves "1.8.0_x".
    if first == 1 {
        parts.next()?.parse().ok()
    } else {
        Some(first)
    }
}

/// The `java` executable inside a JDK/JRE home folder, if there is one.
pub fn java_executable_in(home: &Path) -> Option<PathBuf> {
    let exe = if cfg!(windows) { "java.exe" } else { "java" };
    [
        home.join("bin").join(exe),
        home.join("Contents").join("Home").join("bin").join(exe),
    ]
    .into_iter()
    .find(|p| p.is_file())
}

/// Run `java -version` and report the major version. Fails if it doesn't run in 10 s.
pub async fn probe(java: &Path) -> Result<u32> {
    let mut cmd = tokio::process::Command::new(java);
    cmd.arg("-version").kill_on_drop(true);
    crate::launch::hide_console_window(&mut cmd);
    let output = tokio::time::timeout(Duration::from_secs(10), cmd.output())
        .await
        .map_err(|_| LaunchError::BadJava("it did not answer".into()))?
        .map_err(|e| LaunchError::BadJava(e.kind().to_string()))?;
    let text = format!(
        "{}{}",
        String::from_utf8_lossy(&output.stderr),
        String::from_utf8_lossy(&output.stdout)
    );
    parse_java_major(&text)
        .ok_or_else(|| LaunchError::BadJava("unrecognised version output".into()).into())
}

/// Places a JDK is commonly installed.
fn candidate_executables() -> Vec<PathBuf> {
    let mut homes: Vec<PathBuf> = Vec::new();
    if let Some(home) = std::env::var_os("JAVA_HOME") {
        homes.push(PathBuf::from(home));
    }
    let mut add_children = |dir: &str, suffix: &[&str]| {
        if let Ok(entries) = std::fs::read_dir(dir) {
            for e in entries.flatten() {
                let mut p = e.path();
                p.extend(suffix);
                homes.push(p);
            }
        }
    };
    add_children("/usr/lib/jvm", &[]);
    add_children("/Library/Java/JavaVirtualMachines", &["Contents", "Home"]);
    for base in [
        "C:\\Program Files\\Eclipse Adoptium",
        "C:\\Program Files\\Java",
        "C:\\Program Files\\Microsoft",
        "C:\\Program Files\\Zulu",
    ] {
        add_children(base, &[]);
    }
    let mut found: Vec<PathBuf> = homes.iter().filter_map(|h| java_executable_in(h)).collect();
    if let Some(path) = std::env::var_os("PATH") {
        let exe = if cfg!(windows) { "java.exe" } else { "java" };
        found.extend(
            std::env::split_paths(&path)
                .map(|d| d.join(exe))
                .filter(|p| p.is_file()),
        );
    }
    // The same JDK is often reachable by several routes.
    let mut seen = std::collections::HashSet::new();
    found.retain(|p| seen.insert(p.canonicalize().unwrap_or_else(|_| p.clone())));
    found
}

/// Adoptium's names for this computer, or `None` if it publishes no build for it.
fn adoptium_platform() -> Option<(&'static str, &'static str)> {
    let os = match std::env::consts::OS {
        "linux" => "linux",
        "macos" => "mac",
        "windows" => "windows",
        _ => return None,
    };
    let arch = match std::env::consts::ARCH {
        "x86_64" => "x64",
        "aarch64" => "aarch64",
        _ => return None,
    };
    Some((os, arch))
}

    java_dir: PathBuf,

    downloader: Downloader,
    api_base: String,
}

impl JavaManager {
    pub fn new(db: Db, java_dir: &Path, max_concurrent: u8) -> Result<Self> {
        let ua = format!("{APP_NAME}/{APP_VERSION}");
        Ok(Self::with_clients(
            db,
            java_dir,
            HttpClient::new(&ua, api_policy())?,
            Downloader::new(
                HttpClient::new_for_downloads(&ua, download_policy())?,
                max_concurrent,
            ),
            ADOPTIUM_API.into(),
        ))
    }

    pub(crate) fn with_clients(
        db: Db,
        java_dir: &Path,
        api: HttpClient,
        downloader: Downloader,
        api_base: String,
    ) -> Self {
        Self {
            db,
            java_dir: java_dir.to_path_buf(),
            api,
            downloader,
            api_base,
        }
    }

    /// A `java` executable of exactly this major version: one Kiln already knows about, one
    /// installed on the system, or a fresh Temurin download, in that order.
    pub async fn ensure(
        &self,
        major: u32,
        progress: Option<watch::Sender<Progress>>,
    ) -> Result<PathBuf> {
        if let Some(path) = self.known(major).await? {
            return Ok(path);
        }
        for candidate in candidate_executables() {
            if probe(&candidate).await.is_ok_and(|m| m == major) {
                self.record(&candidate, "System", major, false).await?;
                return Ok(candidate);
            }
        }
        self.install_temurin(major, progress).await
    }

    /// The executable of a recorded runtime, if it is known and still on disk.
    pub async fn runtime_path(&self, id: &str) -> Result<Option<PathBuf>> {
        let row: Option<(String,)> = sqlx::query_as("SELECT path FROM java_runtimes WHERE id = ?1")
            .bind(id)
            .fetch_optional(&self.db)
            .await?;
        Ok(row.map(|(p,)| PathBuf::from(p)).filter(|p| p.is_file()))
    }

    /// A recorded runtime whose file still exists (the user may have uninstalled it).
    async fn known(&self, major: u32) -> Result<Option<PathBuf>> {
        let rows: Vec<(String,)> =
            sqlx::query_as("SELECT path FROM java_runtimes WHERE major = ?1 ORDER BY managed DESC")
                .bind(i64::from(major))
                .fetch_all(&self.db)
                .await?;
        Ok(rows
            .into_iter()
            .map(|(p,)| PathBuf::from(p))
            .find(|p| p.is_file()))
    }

    async fn record(&self, exe: &Path, vendor: &str, major: u32, managed: bool) -> Result<()> {
        let path = exe.to_string_lossy().into_owned();
        let id = hash_bytes(HashAlgo::Sha1, path.as_bytes());
        sqlx::query(
            "INSERT INTO java_runtimes (id, vendor, major, path, managed) VALUES (?1, ?2, ?3, ?4, ?5) \
             ON CONFLICT (path) DO UPDATE SET vendor = excluded.vendor, major = excluded.major, \
             managed = excluded.managed",
        )
        .bind(id)
        .bind(vendor)
        .bind(i64::from(major))
        .bind(path)
        .bind(i64::from(managed))
        .execute(&self.db)
        .await?;
        Ok(())
    }

    async fn find_package(&self, major: u32) -> Result<Package> {
        let (os, arch) = adoptium_platform().ok_or(LaunchError::NoJavaBuild(major))?;
        let url = format!(
            "{}/v3/assets/latest/{major}/hotspot?architecture={arch}&image_type=jre&os={os}&vendor=eclipse",
            self.api_base
        );
        let resp = self
            .api
            .execute(Method::GET, &url, Body::Empty, None)
            .await?;
        if !resp.is_success() {
            return Err(Error::Http(format!(
                "Adoptium answered HTTP {}",
                resp.status
            )));
        }
        let assets: Vec<Asset> = resp.json()?;
        assets
            .into_iter()
            .next()
            .map(|a| a.binary.package)
            .ok_or_else(|| LaunchError::NoJavaBuild(major).into())
    }

    async fn install_temurin(
        &self,
        major: u32,
        progress: Option<watch::Sender<Progress>>,
    ) -> Result<PathBuf> {
        let package = self.find_package(major).await?;
        if package.name.contains(['/', '\\']) {
            return Err(Error::InvalidPath("package name".into()));
        }
        let archive = SafePath::new(&self.java_dir, &format!("downloads/{}", package.name))?;
        self.downloader
            .download_all(
                vec![DownloadTask {
                    url: package.link,
                    dest: archive.clone(),
                    hash: Some(ExpectedHash::new(HashAlgo::Sha256, &package.checksum)?),
                    size: Some(package.size),
                }],
                progress,
            )
            .await?;

        let staging = self.java_dir.join(format!("{major}.tmp"));
        let target = self.java_dir.join(major.to_string());
        let (archive_path, staging_path) = (archive.as_path().to_path_buf(), staging.clone());
        let name = package.name.clone();
        tokio::task::spawn_blocking(move || -> Result<()> {
            let _ = std::fs::remove_dir_all(&staging_path);
            let opts = ExtractOptions {
                strip_components: 1,
                ..Default::default()
            };
            if name.ends_with(".zip") {
                extract_zip(&archive_path, &staging_path, &opts)?;
            } else if name.ends_with(".tar.gz") || name.ends_with(".tgz") {
                extract_tar_gz(&archive_path, &staging_path, &opts)?;
            } else {
                return Err(Error::Archive("unknown package format".into()));
            }
            Ok(())
        })
        .await
        .map_err(|e| Error::Io(std::io::Error::other(e)))??;

        // Check the unpacked runtime before it replaces anything.
        let staged_exe = java_executable_in(&staging)
            .ok_or_else(|| LaunchError::BadJava("the package has no java executable".into()))?;
        let actual = probe(&staged_exe).await?;
        if actual != major {
            let _ = std::fs::remove_dir_all(&staging);
            return Err(
                LaunchError::BadJava(format!("expected Java {major}, got {actual}")).into(),
            );
        }
        let _ = std::fs::remove_dir_all(&target);
        std::fs::rename(&staging, &target)?;
        let _ = std::fs::remove_file(archive.as_path());

        let exe = java_executable_in(&target)
            .ok_or_else(|| LaunchError::BadJava("the java executable went missing".into()))?;
        self.record(&exe, "Eclipse Temurin", major, true).await?;
        tracing::info!(major, "installed a Java runtime");
        Ok(exe)
    }
}

//! Mojang's launcher metadata: the version manifest, per-version JSON, asset indexes, and
//! the download tasks for the client jar, libraries and assets.
//!
//! Everything fetched here is verified against the SHA-1 Mojang publishes and cached under
//! `cache/{meta,versions,libraries,assets,jars}`.

pub mod rules;
pub mod version;

use std::path::{Path, PathBuf};

use reqwest::Method;
use serde::{Deserialize, Serialize};

use crate::download::{DownloadTask, Downloader};
use crate::hash::{ExpectedHash, HashAlgo};
use crate::launch::LaunchError;
use crate::net::{Body, HostPolicy, HttpClient};
use crate::paths::SafePath;
use crate::{Error, Result, APP_NAME, APP_VERSION};

pub use version::{AssetIndex, LibraryFile, ResolvedLibraries, VersionJson};

pub const MANIFEST_URL: &str = "https://piston-meta.mojang.com/mc/game/version_manifest_v2.json";
pub const ASSET_BASE: &str = "https://resources.download.minecraft.net/";
/// Refuse version JSON files larger than this (real ones are well under 1 MiB).
const MAX_JSON_BYTES: u64 = 8 * 1024 * 1024;

/// The only hosts the Mojang clients may contact.
pub fn host_policy() -> HostPolicy {
    HostPolicy::new(&[
        "piston-meta.mojang.com",
        "piston-data.mojang.com",
        "launchermeta.mojang.com",
        "launcher.mojang.com",
        "libraries.minecraft.net",
        "resources.download.minecraft.net",
    ])
}

/// Version ids become folder and file names, so keep them boring.
pub fn is_valid_version_id(id: &str) -> bool {
    (1..=64).contains(&id.len())
        && !id.starts_with('.')
        && id
            .bytes()
            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-' | b'+'))
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Latest {
    pub release: String,
    pub snapshot: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct VersionEntry {
    pub id: String,
    #[serde(rename = "type")]
    pub kind: String,
    pub url: String,
    pub sha1: String,
    #[serde(rename = "releaseTime")]
    pub release_time: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct VersionManifest {
    pub latest: Latest,
    pub versions: Vec<VersionEntry>,
}

/// What the UI is told about a version. No URLs or hashes cross IPC.
#[derive(Debug, Clone, Serialize, PartialEq, Eq)]
pub struct VersionSummary {
    pub id: String,
    pub kind: String,
    pub release_time: String,
}

impl VersionManifest {
    pub fn summaries(&self) -> Vec<VersionSummary> {
        self.versions
            .iter()
            .filter(|v| is_valid_version_id(&v.id))
            .map(|v| VersionSummary {
                id: v.id.clone(),
                kind: v.kind.clone(),
                release_time: v.release_time.clone(),
            })
            .collect()
    }
}

#[derive(Debug, Clone)]
pub struct MojangClient {
    api: HttpClient,
    downloader: Downloader,
    cache: PathBuf,
    manifest_url: String,
    asset_base: String,
}

fn sha1_of(hex: &str) -> Result<ExpectedHash> {
    ExpectedHash::new(HashAlgo::Sha1, hex)
}

impl MojangClient {
    pub fn new(cache: &Path, max_concurrent: u8) -> Result<Self> {
        let ua = format!("{APP_NAME}/{APP_VERSION}");
        Ok(Self::with_clients(
            HttpClient::new(&ua, host_policy())?,
            Downloader::new(
                HttpClient::new_for_downloads(&ua, host_policy())?,
                max_concurrent,
            ),
            cache,
            MANIFEST_URL.into(),
            ASSET_BASE.into(),
        ))
    }

    pub(crate) fn with_clients(
        api: HttpClient,
        downloader: Downloader,
        cache: &Path,
        manifest_url: String,
        asset_base: String,
    ) -> Self {
        Self {
            api,
            downloader,
            cache: cache.to_path_buf(),
            manifest_url,
            asset_base,
        }
    }

    pub fn downloader(&self) -> &Downloader {
        &self.downloader
    }

    /// A folder under the cache, created on demand (`SafePath` needs its root to exist).
    pub fn cache_dir(&self, name: &str) -> Result<PathBuf> {
        let dir = self.cache.join(name);
        std::fs::create_dir_all(&dir)?;
        Ok(dir)
    }

    fn task(
        &self,
        folder: &str,
        rel: &str,
        url: &str,
        sha1: &str,
        size: Option<u64>,
    ) -> Result<DownloadTask> {
        Ok(DownloadTask {
            url: url.to_owned(),
            dest: SafePath::new(&self.cache_dir(folder)?, rel)?,
            hash: Some(sha1_of(sha1)?),
            size,
        })
    }

    /// The version list. Falls back to the copy saved by the last successful fetch when
    /// Mojang can't be reached, so installed versions still launch offline.
    pub async fn manifest(&self) -> Result<VersionManifest> {
        let saved = self.cache_dir("meta")?.join("version_manifest_v2.json");
        let fetched = match self
            .api
            .execute(Method::GET, &self.manifest_url, Body::Empty, None)
            .await
        {
            Ok(resp) if resp.is_success() => resp.json::<VersionManifest>(),
            Ok(resp) => Err(Error::Http(format!("Mojang answered HTTP {}", resp.status))),
            Err(e) => Err(e),
        };
        match fetched {
            Ok(manifest) => {
                let tmp = saved.with_extension("json.tmp");
                std::fs::write(&tmp, serde_json::to_vec(&manifest)?)?;
                std::fs::rename(&tmp, &saved)?;
                Ok(manifest)
            }
            Err(e) => match std::fs::read(&saved) {
                Ok(bytes) => {
                    tracing::warn!("using the saved version list: Mojang could not be reached");
                    Ok(serde_json::from_slice(&bytes)?)
                }
                Err(_) => Err(e),
            },
        }
    }

    pub async fn version_json(&self, id: &str) -> Result<VersionJson> {
        if !is_valid_version_id(id) {
            return Err(LaunchError::InvalidVersionId.into());
        }
        let dest = SafePath::new(&self.cache_dir("versions")?, &format!("{id}/{id}.json"))?;
        match self.manifest().await {
            Ok(manifest) => {
                let entry = manifest
                    .versions
                    .iter()
                    .find(|v| v.id == id)
                    .ok_or_else(|| LaunchError::UnknownVersion(id.to_owned()))?;
                let task = DownloadTask {
                    url: entry.url.clone(),
                    dest: dest.clone(),
                    hash: Some(sha1_of(&entry.sha1)?),
                    size: None,
                };
                self.downloader.download(task).await?;
            }
            // Offline: a previously verified copy is still good.
            Err(e) if !dest.as_path().is_file() => return Err(e),
            Err(_) => {}
        }
        read_json(dest.as_path())
    }

    pub async fn asset_index(&self, version: &VersionJson) -> Result<AssetIndex> {
        let r = version.asset_index_ref()?;
        if !is_valid_version_id(&r.id) {
            return Err(LaunchError::InvalidVersionId.into());
        }
        let task = self.task(
            "assets",
            &format!("indexes/{}.json", r.id),
            &r.url,
            &r.sha1,
            Some(r.size),
        )?;
        let dest = task.dest.clone();
        self.downloader.download(task).await?;
        let index: AssetIndex = read_json(dest.as_path())?;
        if index.is_virtual || index.map_to_resources {
            return Err(LaunchError::LegacyAssets(version.id.clone()).into());
        }
        Ok(index)
    }

    pub fn asset_tasks(&self, index: &AssetIndex) -> Result<Vec<DownloadTask>> {
        index
            .objects
            .values()
            .map(|o| {
                let hash = sha1_of(&o.hash)?;
                let prefix = &hash.hex()[..2];
                Ok(DownloadTask {
                    url: format!("{}{prefix}/{}", self.asset_base, hash.hex()),
                    dest: SafePath::new(
                        &self.cache_dir("assets")?,
                        &format!("objects/{prefix}/{}", hash.hex()),
                    )?,
                    size: Some(o.size),
                    hash: Some(hash),
                })
            })
            .collect()
    }

    /// Where a library lives (or will live) in the shared libraries cache.
    pub fn library_path(&self, lib: &LibraryFile) -> Result<SafePath> {
        SafePath::new(&self.cache_dir("libraries")?, &lib.path)
    }

    pub fn library_task(&self, lib: &LibraryFile) -> Result<DownloadTask> {
        self.task("libraries", &lib.path, &lib.url, &lib.sha1, lib.size)
    }

    /// Where the client jar lives once downloaded, and the task to fetch it.
    pub fn client_jar_task(&self, version: &VersionJson) -> Result<DownloadTask> {
        let client = &version
            .downloads
            .as_ref()
            .ok_or_else(|| LaunchError::UnsupportedLibrary(format!("{} client jar", version.id)))?
            .client;
        self.task(
            "jars",
            &format!("{0}/{0}.jar", version.id),
            &client.url,
            &client.sha1,
            client.size,
        )
    }

    pub fn assets_root(&self) -> Result<PathBuf> {
        self.cache_dir("assets")
    }
}

fn read_json<T: serde::de::DeserializeOwned>(path: &Path) -> Result<T> {
    if std::fs::metadata(path)?.len() > MAX_JSON_BYTES {
        return Err(Error::Json(
            <serde_json::Error as serde::de::Error>::custom("metadata file is too large"),
        ));
    }
    Ok(serde_json::from_slice(&std::fs::read(path)?)?)
}

//! Mojang's `rules` mechanism: which libraries and arguments apply to this computer.
//!
//! A list with no rules always applies. Otherwise it starts as "not allowed" and each rule
//! that matches this environment sets the outcome to its action, so the last match wins.

use std::collections::{HashMap, HashSet};

use regex::RegexBuilder;
use serde::Deserialize;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Os {
    Windows,
    MacOs,
    Linux,
}

impl Os {
    pub fn current() -> Self {
        if cfg!(target_os = "windows") {
            Self::Windows
        } else if cfg!(target_os = "macos") {
            Self::MacOs
        } else {
            Self::Linux
        }
    }

    /// The name Mojang's manifests use.
    pub fn mojang_name(self) -> &'static str {
        match self {
            Self::Windows => "windows",
            Self::MacOs => "osx",
            Self::Linux => "linux",
        }
    }
}

/// What the rules are evaluated against.
#[derive(Debug, Clone)]
pub struct Environment {
    pub os: Os,
    /// `x86` (32-bit), `x86_64` or `arm64`: the spellings Mojang's manifests use.
    pub arch: String,
    /// Matched by the `os.version` regex. Windows reports `10.0` for both 10 and 11; other
    /// systems are left unset, so version-restricted rules don't match there.
    pub os_version: Option<String>,
    /// Launcher features switched on (`has_custom_resolution`, `is_demo_user`, ...).
    pub features: HashSet<String>,
}

impl Environment {
    pub fn current() -> Self {
        let os = Os::current();
        let arch = match std::env::consts::ARCH {
            "aarch64" => "arm64",
            "x86" => "x86",
            _ => "x86_64",
        };
        Self {
            os,
            arch: arch.into(),
            os_version: (os == Os::Windows).then(|| "10.0".into()),
            features: HashSet::new(),
        }
    }

    /// Value substituted for `${arch}` in old-style native classifiers.
    pub fn bits(&self) -> &'static str {
        if self.arch == "x86" {
            "32"
        } else {
            "64"
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum Action {
    Allow,
    Disallow,
}

#[derive(Debug, Clone, Default, Deserialize)]
pub struct OsRule {
    pub name: Option<String>,
    pub arch: Option<String>,
    pub version: Option<String>,
}

#[derive(Debug, Clone, Deserialize)]
pub struct Rule {
    pub action: Action,
    #[serde(default)]
    pub os: Option<OsRule>,
    #[serde(default)]
    pub features: Option<HashMap<String, bool>>,
}

impl OsRule {
    fn matches(&self, env: &Environment) -> bool {
        if self
            .name
            .as_deref()
            .is_some_and(|n| n != env.os.mojang_name())
        {
            return false;
        }
        if self.arch.as_deref().is_some_and(|a| a != env.arch) {
            return false;
        }
        match (&self.version, &env.os_version) {
            (None, _) => true,
            (Some(_), None) => false,
            // The pattern comes from a downloaded file: cap its compiled size.
            (Some(pattern), Some(version)) => RegexBuilder::new(pattern)
                .size_limit(100_000)
                .build()
                .is_ok_and(|re| re.is_match(version)),
        }
    }
}

impl Rule {
    fn matches(&self, env: &Environment) -> bool {
        self.os.as_ref().is_none_or(|os| os.matches(env))
            && self
                .features
                .as_ref()
                .is_none_or(|f| f.iter().all(|(k, v)| env.features.contains(k) == *v))
    }
}

pub fn rules_allow(rules: &[Rule], env: &Environment) -> bool {
    if rules.is_empty() {
        return true;
    }
    let mut allowed = false;
    for rule in rules.iter().filter(|r| r.matches(env)) {
        allowed = rule.action == Action::Allow;
    }
    allowed
}

impl Argument {
    /// The argument strings this entry contributes in `env` (none if its rules say no).
    pub fn values(&self, env: &Environment) -> Vec<&str> {
        match self {
            Self::Plain(s) => vec![s.as_str()],
            Self::Conditional { rules, value } if rules_allow(rules, env) => match value {
                OneOrMany::One(s) => vec![s.as_str()],
                OneOrMany::Many(v) => v.iter().map(String::as_str).collect(),

fn to_file(name: &str, f: &FileRef) -> Result<LibraryFile> {
    let path = f
        .path
        .clone()
        .ok_or_else(|| LaunchError::UnsupportedLibrary(name.to_owned()))?;
    Ok(LibraryFile {
        path,
        url: f.url.clone(),
        sha1: f.sha1.clone(),
        size: f.size,
    })
}

impl VersionJson {
    pub fn asset_index_ref(&self) -> Result<&AssetIndexRef> {
        self.asset_index.as_ref().ok_or_else(|| {
            LaunchError::UnsupportedLibrary(format!("{} (no asset index)", self.id)).into()
        })
    }

    /// Combine a loader profile (`self`, which names its parent in `inheritsFrom`) with the
    /// vanilla version it builds on. The loader's main class wins, argument lists are
    /// concatenated parent first, and the loader's libraries come first on the classpath
    /// (replacing any parent library with the same `group:artifact`).
    pub fn merged_onto(&self, parent: &VersionJson) -> VersionJson {
        let ga = |name: &str| name.splitn(3, ':').take(2).collect::<Vec<_>>().join(":");
        let child_keys: std::collections::HashSet<String> =
            self.libraries.iter().map(|l| ga(&l.name)).collect();
        let mut libraries = self.libraries.clone();
        libraries.extend(
            parent
                .libraries
                .iter()
                .filter(|l| !child_keys.contains(&ga(&l.name)))
                .cloned(),
        );
        let arguments = match (&parent.arguments, &self.arguments) {
            (Some(p), Some(c)) => Some(Arguments {
                jvm: p.jvm.iter().chain(&c.jvm).cloned().collect(),
                game: p.game.iter().chain(&c.game).cloned().collect(),
            }),
            (Some(p), None) => Some(p.clone()),
            (None, Some(c)) => Some(c.clone()),
            (None, None) => None,
        };
        VersionJson {
            id: self.id.clone(),
            kind: self.kind.clone(),
            main_class: self.main_class.clone(),
            minecraft_arguments: if arguments.is_some() {
                None
            } else {
                self.minecraft_arguments
                    .clone()
                    .or_else(|| parent.minecraft_arguments.clone())
            },
            arguments,
            asset_index: self
                .asset_index
                .clone()
                .or_else(|| parent.asset_index.clone()),
            downloads: self.downloads.clone().or_else(|| parent.downloads.clone()),
            libraries,
            java_version: self
                .java_version
                .clone()
                .or_else(|| parent.java_version.clone()),
            inherits_from: None,
        }
    }

    /// Java major version the game needs. Versions without the field predate it: Java 8.
    pub fn java_major(&self) -> u32 {
        self.java_version.as_ref().map_or(8, |j| j.major_version)
    }

    pub fn resolve_libraries(&self, env: &Environment) -> Result<ResolvedLibraries> {
        let mut out = ResolvedLibraries::default();
        for lib in &self.libraries {
            if !rules_allow(&lib.rules, env) {
                continue;
            }
            let downloads = lib.downloads.as_ref();
            let classifier = lib.natives.as_ref().and_then(|n| {
                n.get(env.os.mojang_name())
                    .map(|c| c.replace("${arch}", env.bits()))
            });
            if let Some(classifier) = &classifier {
                // A natives library for another platform just doesn't apply here.
                if let Some(file) = downloads
                    .and_then(|d| d.classifiers.as_ref())
                    .and_then(|c| c.get(classifier))
                {
                    let exclude = lib.extract.clone().unwrap_or_default().exclude;
                    out.natives.push((to_file(&lib.name, file)?, exclude));
                }
            }
            match downloads.and_then(|d| d.artifact.as_ref()) {
                Some(artifact) => {
                    let file = to_file(&lib.name, artifact)?;
                    if !out.classpath.iter().any(|f| f.path == file.path) {
                        out.classpath.push(file);
                    }
                }
                // A library with natives only has no jar of its own for the classpath.
                None if lib.natives.is_some() => {}
                None => return Err(LaunchError::UnsupportedLibrary(lib.name.clone()).into()),
            }
        }
        Ok(out)
    }
}

/// `group:artifact:version[:classifier][@ext]` to its Maven repository path.
pub fn maven_path(name: &str) -> Option<String> {
    let (coords, ext) = name.split_once('@').unwrap_or((name, "jar"));
    let parts: Vec<&str> = coords.split(':').collect();
    if !(3..=4).contains(&parts.len()) || parts.iter().any(|p| p.is_empty()) || ext.is_empty() {
        return None;
    }
    let (group, artifact, version) = (parts[0].replace('.', "/"), parts[1], parts[2]);
    let file = match parts.get(3) {
        Some(classifier) => format!("{artifact}-{version}-{classifier}.{ext}"),
        None => format!("{artifact}-{version}.{ext}"),
    };
    Some(format!("{group}/{artifact}/{version}/{file}"))
}

        let jvm: Vec<_> = args.jvm.iter().flat_map(|a| a.values(&linux)).collect();
        assert_eq!(jvm, ["-cp", "${classpath}"]);
        let mac: Vec<_> = args
            .jvm
            .iter()
            .flat_map(|a| a.values(&env(Os::MacOs)))
            .collect();
        assert_eq!(mac[0], "-XstartOnFirstThread");
    }

    #[test]
    fn a_loader_profile_merges_onto_its_parent() {
        let parent = version(vec![
            lib(
                json!({"name": "org.ow2.asm:asm:9.3", "downloads": {"artifact": art("org/ow2/asm/asm/9.3/asm-9.3.jar")}}),
            ),
            lib(
                json!({"name": "a:other:1", "downloads": {"artifact": art("a/other/1/other-1.jar")}}),
            ),
        ]);
        let parent: VersionJson = {
            let mut p = parent;
            p.arguments = Some(Arguments {
                jvm: vec![Argument::Plain("-Pjvm".into())],
                game: vec![Argument::Plain("--pgame".into())],
            });
            p
        };
        let child: VersionJson = serde_json::from_value(json!({
            "id": "fabric-loader-0.16.5-1.21.1", "type": "release", "inheritsFrom": "1.21.1",
            "mainClass": "net.fabricmc.loader.impl.launch.knot.KnotClient",
            "arguments": {"jvm": ["-Cjvm"], "game": []},
            "libraries": [{"name": "org.ow2.asm:asm:9.7", "downloads": {"artifact": art("org/ow2/asm/asm/9.7/asm-9.7.jar")}},
                          {"name": "net.fabricmc:fabric-loader:0.16.5", "downloads": {"artifact": art("net/fabricmc/fabric-loader/0.16.5/fabric-loader-0.16.5.jar")}}]
        })).unwrap();
        let m = child.merged_onto(&parent);
        assert_eq!(m.id, "fabric-loader-0.16.5-1.21.1");
        assert_eq!(
            m.main_class,
            "net.fabricmc.loader.impl.launch.knot.KnotClient"
        );
        assert!(m.inherits_from.is_none());
        assert_eq!(
            m.asset_index.as_ref().unwrap().id,
            "17",
            "inherited from the parent"
        );
        let names: Vec<_> = m.libraries.iter().map(|l| l.name.as_str()).collect();
        assert_eq!(
            names,
            [
                "org.ow2.asm:asm:9.7",
                "net.fabricmc:fabric-loader:0.16.5",
                "a:other:1"
            ],
            "child first, same group:artifact replaced"
        );
        let a = m.arguments.unwrap();
        let env = env(Os::Linux);
        assert_eq!(
            a.jvm
                .iter()
                .flat_map(|x| x.values(&env))
                .collect::<Vec<_>>(),
            ["-Pjvm", "-Cjvm"]
        );
        assert_eq!(
            a.game
                .iter()
                .flat_map(|x| x.values(&env))
                .collect::<Vec<_>>(),
            ["--pgame"]
        );
    }
