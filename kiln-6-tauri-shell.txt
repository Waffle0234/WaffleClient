//! Sign-in commands. None of these ever returns a token: the UI only sees accounts and the
//! short code the user types on Microsoft's page.

use kiln_core::auth::{is_valid_account_id, Account, AuthError, DeviceCodeInfo};
use kiln_core::{AppError, Error};
use tauri::State;

use crate::state::AppState;

fn check_id(id: &str) -> Result<(), AppError> {
    if is_valid_account_id(id) {
        Ok(())
    } else {
        Err(Error::from(AuthError::InvalidAccountId).into())
    }
}

#[tauri::command]

    state: State<'_, AppState>,

    Ok(state.auth.begin_device_flow().await?)
}

/// `auth_begin_device_flow`. Can take minutes; `auth_cancel` ends it early.

    Ok(state.auth.poll().await?)

/// Opens the sign-in page in the default browser. Takes no arguments on purpose: the
/// address comes from the validated pending flow, never from the webview.
#[tauri::command]
pub async fn auth_open_verification_page(state: State<'_, AppState>) -> Result<(), AppError> {
    let uri = state
        .auth
        .pending_verification_uri()
        .ok_or_else(|| Error::from(AuthError::NoPendingFlow))?;
    tauri::async_runtime::spawn_blocking(move || opener::open(uri))
        .await
        .map_err(|e| Error::Io(std::io::Error::other(e)))?
        .map_err(|e| Error::Io(std::io::Error::other(e.to_string())))?;
    Ok(())
}

    Ok(state.auth.list_accounts().await?)
}

//! Instance management commands. The webview never supplies a file path: imports name a
//! file inside Kiln's `imports/` folder and exports are written to `exports/`.

use kiln_core::instance::pack::{ImportFile, PackService};
use kiln_core::instance::{Instance, InstanceSettings, NewInstance};
use kiln_core::loader::{LoaderClient, LoaderKind, LoaderVersion};
use kiln_core::mods::service::ModService;
use kiln_core::mojang::{MojangClient, VersionSummary};
use kiln_core::{AppError, Error};
use serde::Deserialize;
use tauri::State;

use crate::state::AppState;

fn parse_loader(loader: &str) -> Result<LoaderKind, AppError> {
    LoaderKind::parse(loader).ok_or_else(|| {
        Error::InvalidSetting {
            key: "loader".into(),
            reason: "choose Vanilla, Fabric, Quilt, NeoForge or Forge".into(),
        }
        .into()
    })
}

async fn clients(state: &AppState) -> Result<(MojangClient, LoaderClient), AppError> {
    let max = state.settings.load().await?.max_concurrent_downloads;
    let mojang = MojangClient::new(&state.dirs.cache, max)?;
    let loaders = LoaderClient::new(mojang.clone(), max)?;
    Ok((mojang, loaders))
}

/// Minecraft versions that work with `loader`, newest first. Versions the loader supports
/// but Mojang no longer lists are left out, so every choice can actually be launched.
#[tauri::command]
pub async fn picker_game_versions(
    state: State<'_, AppState>,
    loader: String,
) -> Result<Vec<VersionSummary>, AppError> {
    let kind = parse_loader(&loader)?;
    let (mojang, loaders) = clients(&state).await?;
    let all = mojang.manifest().await?.summaries();
    if kind == LoaderKind::Vanilla {
        return Ok(all);
    }
    let supported: std::collections::HashSet<String> =
        loaders.game_versions(kind).await?.into_iter().collect();
    Ok(all
        .into_iter()
        .filter(|v| supported.contains(&v.id))
        .collect())
}

#[tauri::command]
pub async fn picker_loader_versions(
    state: State<'_, AppState>,
    loader: String,
    mc_version: String,
) -> Result<Vec<LoaderVersion>, AppError> {
    let kind = parse_loader(&loader)?;
    let (_, loaders) = clients(&state).await?;
    Ok(loaders.loader_versions(kind, &mc_version).await?)
}

#[derive(Debug, serde::Serialize)]
pub struct ExportResult {
    pub file: String,
    /// Mods listed by Modrinth hash instead of being copied into the pack.
    pub listed: usize,
    /// Jars Modrinth doesn't know, which the pack bundles. The person should check they may
    /// share them.
    pub bundled_unverified: Vec<String>,
}

/// Writes the pack to the exports folder and returns its file name.
#[tauri::command]
pub async fn instance_export(
    state: State<'_, AppState>,
    id: String,
    format: ExportFormat,
) -> Result<ExportResult, AppError> {
    Ok(match format {
        ExportFormat::Kiln => ExportResult {
            file: packs(&state).await?.export_native(&id).await?,
            listed: 0,
            bundled_unverified: Vec::new(),
        },
        ExportFormat::Mrpack => {
            let _ops = state.mod_ops.lock().await;
            let max = state.settings.load().await?.max_concurrent_downloads;
            let svc = ModService::new(
                state.modrinth.clone(),
                state.mods.clone(),
                packs(&state).await?,
                &state.dirs,
                max,
            )?;
            let out = svc.export_mrpack(&id).await?;
            ExportResult {
                file: out.file,
                listed: out.listed,
                bundled_unverified: out.bundled_unverified,
            }
        }
    })
}

/// Open one of Kiln's folders in the system file manager. Only fixed choices are accepted.
#[tauri::command]
pub async fn open_folder(
    state: State<'_, AppState>,
    which: String,
    id: Option<String>,
) -> Result<(), AppError> {
    let path = match (which.as_str(), id) {
        ("imports", _) => state.dirs.imports.clone(),
        ("exports", _) => state.dirs.exports.clone(),
        ("instance", Some(id)) => state.instances.game_dir(&id).await?.as_path().to_path_buf(),
        _ => {
            return Err(Error::InvalidSetting {
                key: "folder".into(),
                reason: "choose imports, exports or an instance".into(),
            }
            .into())
        }
    };
    tauri::async_runtime::spawn_blocking(move || opener::open(path))
        .await
        .map_err(|e| Error::Io(std::io::Error::other(e)))?
        .map_err(|e| Error::Io(std::io::Error::other(e.to_string())))?;
    Ok(())
}

//! Game launching commands. Progress and output reach the UI as events (`launch-status`,
//! `game-log`); no command or event ever carries a token.

use std::sync::Arc;

use kiln_core::instance::parse_jvm_args;
use kiln_core::java::JavaManager;
use kiln_core::launch::{
    LaunchDeps, LaunchError, LaunchEvents, LaunchRequest, LaunchStatus, LogLine,
};
use kiln_core::loader::LoaderClient;
use kiln_core::mods::store::RollbackReport;
use kiln_core::mojang::MojangClient;
use kiln_core::{AppError, Error};
use tauri::{AppHandle, Emitter, State};

use crate::state::AppState;

struct TauriEvents(AppHandle);

impl LaunchEvents for TauriEvents {
    fn status(&self, status: &LaunchStatus) {
        if let Err(e) = self.0.emit("launch-status", status) {
            tracing::debug!(error = %e, "could not emit launch-status");
        }
    }

    fn log(&self, line: &LogLine) {
        // Lines were redacted in kiln-core before they got here.
        if let Err(e) = self.0.emit("game-log", line) {
            tracing::debug!(error = %e, "could not emit game-log");
        }
    }

    fn rolled_back(&self, report: &RollbackReport) {
        if let Err(e) = self.0.emit("mods-rolled-back", report) {
            tracing::debug!(error = %e, "could not emit mods-rolled-back");
        }
    }
}

/// Start the game from a stored instance, using the active account.
#[tauri::command]
pub async fn launch_start(
    app: AppHandle,
    state: State<'_, AppState>,
    instance_id: String,
) -> Result<(), AppError> {
    if state.launch.is_active() {
        return Err(Error::from(LaunchError::AlreadyRunning).into());
    }
    // Don't start the game while mods or a modpack are being changed.
    let _no_mod_changes = state
        .mod_ops
        .try_lock()
        .map_err(|_| Error::from(LaunchError::InstanceBusy))?;
    let instance = state.instances.get(&instance_id).await?;
    let game_dir = state.instances.game_dir(&instance_id).await?;
    let account = state
        .auth
        .list_accounts()
        .await?
        .into_iter()
        .find(|a| a.is_active)
        .ok_or_else(|| Error::from(LaunchError::NoAccount))?;
    let settings = state.settings.load().await?;
    let max = settings.max_concurrent_downloads;
    // Refreshes the Microsoft sign-in if needed; the session never leaves the backend.
    let session = state.auth.session_for_launch(&account.id).await?;

    let mojang = MojangClient::new(&state.dirs.cache, max)?;
    let java = JavaManager::new(state.db.clone(), &state.dirs.java, max)?;
    let java_path = match &instance.java_id {
        Some(id) => java.runtime_path(id).await?,
        None => None,
    };
    let jvm_args = match instance.jvm_args.as_deref() {
        Some(args) => parse_jvm_args(args)?,
        None => Vec::new(),
    };
    let deps = LaunchDeps {
        dirs: state.dirs.clone(),
        loaders: LoaderClient::new(mojang.clone(), max)?,
        mojang,
        java,
        instances: Some(state.instances.clone()),
        mods: Some(state.mods.clone()),
    };
    let request = LaunchRequest {
        mc_version: instance.mc_version,
        loader: instance.loader,
        loader_version: instance.loader_version,
        game_dir,
        instance_id: Some(instance.id),
        ram_mb: instance.ram_mb.or(settings.default_ram_mb),
        jvm_args,
        java_path,
    };
    state
        .launch
        .start(deps, request, session, Arc::new(TauriEvents(app)))?;
    Ok(())
}

//! Mod browsing, installing, updating and modpack commands. Everything is validated in
//! `kiln-core`; the webview only ever names a project, an instance or a mod row, never a
//! file path or a URL.

use kiln_core::instance::pack::PackService;
use kiln_core::instance::Instance;
use kiln_core::launch::LaunchError;
use kiln_core::mods::modrinth::Modrinth;
use kiln_core::mods::service::{
    InstallPlan, ModService, PackUpdateInfo, PackUpdateOutcome, UpdateAllReport, UpdateInfo,
};
use kiln_core::mods::source::{
    ModSource, ModVersion, Project, SearchPage, SearchQuery, VersionFilter,
};
use kiln_core::mods::store::{ModEntry, RollbackReport};
use kiln_core::{AppError, Error};
use tauri::{AppHandle, State};

async fn service(state: &AppState) -> Result<ModService<Modrinth>, AppError> {
    let max = state.settings.load().await?.max_concurrent_downloads;
    let packs = PackService::new(state.instances.clone(), &state.dirs, max)?;
    Ok(ModService::new(
        state.modrinth.clone(),
        state.mods.clone(),
        packs,
        &state.dirs,
        max,
    )?)
}

/// Mods can't change under a running (or starting) game.
fn ensure_idle(state: &AppState, instance_id: &str) -> Result<(), AppError> {
    if state.launch.running_instance().as_deref() == Some(instance_id) {
        return Err(Error::from(LaunchError::InstanceBusy).into());
    }
    Ok(())
}

//! Progress for long operations (imports, installs, updates) reaches the UI as a
//! `task-progress` event. It carries byte and file counts only: no paths or URLs.

use kiln_core::download::Progress;
use serde::Serialize;
use tauri::{AppHandle, Emitter};
use tokio::sync::watch;

#[derive(Debug, Clone, Serialize)]
struct TaskProgress {
    task: &'static str,
    progress: Progress,
}

/// A sender for the core to report into. A background task forwards each change to the UI
/// and ends when the sender is dropped, so nothing keeps running when idle.
pub fn progress_sender(app: &AppHandle, task: &'static str) -> watch::Sender<Progress> {
    let (tx, mut rx) = watch::channel(Progress::default());
    let app = app.clone();
    tauri::async_runtime::spawn(async move {
        while rx.changed().await.is_ok() {
            let progress = *rx.borrow_and_update();
            if let Err(e) = app.emit("task-progress", TaskProgress { task, progress }) {
                tracing::debug!(error = %e, "could not emit task-progress");
            }
        }
    });
    tx
}

use std::path::Path;
use std::sync::Arc;

use kiln_core::auth::{client_id_from_env, AuthService, KeyringStore};
use kiln_core::db::{self, Db};
use kiln_core::instance::InstanceStore;
use kiln_core::launch::LaunchManager;
use kiln_core::logging::{self, LogGuard};
use kiln_core::mods::cache::HttpCache;
use kiln_core::mods::modrinth::Modrinth;
use kiln_core::mods::store::ModStore;
use kiln_core::paths::DataDirs;
use kiln_core::settings::SettingsStore;

/// Keychain service name. Entries show up under this name in Keychain Access, Credential
/// Manager, and Seahorse/KWallet.
const KEYCHAIN_SERVICE: &str = "app.kiln.launcher";

pub struct AppState {
    pub dirs: DataDirs,
    pub db: Db,
    pub settings: SettingsStore,
    pub auth: Arc<AuthService>,
    pub launch: LaunchManager,
    pub instances: InstanceStore,
    pub mods: ModStore,
    pub modrinth: Modrinth,
    /// Held while a mod or modpack operation runs, so two never change a folder at once.
    pub mod_ops: tokio::sync::Mutex<()>,
    pub log: LogGuard,
}

impl AppState {
    pub async fn init(exe_dir: Option<&Path>) -> kiln_core::Result<Self> {
        let dirs = DataDirs::resolve(exe_dir)?;
        dirs.ensure_created()?;

        // Logging first so database problems are recorded. The saved level is applied
        // as soon as settings are loaded. No background work happens after this.
        let log = logging::init(&dirs.logs, "info", cfg!(debug_assertions))?;
        tracing::info!(
            version = kiln_core::APP_VERSION,
            portable = dirs.portable,
            "starting"
        );

        let db = db::open(&dirs.db_file).await?;
        let settings = SettingsStore::new(db.clone());
        let instances = InstanceStore::new(db.clone(), &dirs.instances);
        log.set_level(&settings.load().await?.log_level)?;
        let mods = ModStore::new(db.clone(), instances.clone(), &dirs.backups);
        let cache = HttpCache::open(db.clone(), &dirs.cache.join("http")).await?;
        let modrinth = Modrinth::new(Some(cache))?;

        let client_id = client_id_from_env();
        if client_id.is_none() {
            tracing::warn!("no Microsoft client id configured: sign-in is disabled");
        }
        let auth = Arc::new(AuthService::new(
            db.clone(),
            Arc::new(KeyringStore::new(KEYCHAIN_SERVICE)),
            client_id,
        )?);
        auth.init().await?;

        Ok(Self {
            dirs,
            db,
            settings,
            auth,
            launch: LaunchManager::new(),
            instances,
            mods,
            modrinth,
            mod_ops: tokio::sync::Mutex::new(()),
            log,
        })
    }
}
