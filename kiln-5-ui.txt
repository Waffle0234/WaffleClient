import { api, errorCode, errorMessage, type Account, type DeviceCodeInfo } from './api';

export type Flow =
  | { phase: 'idle' }
  | { phase: 'starting' }
  | { phase: 'waiting'; code: DeviceCodeInfo; expiresAt: number };

class AccountsStore {
  list = $state<Account[]>([]);
  loaded = $state(false);
  flow = $state<Flow>({ phase: 'idle' });
  error = $state<string | null>(null);

  async load(): Promise<void> {
    try {
      this.list = await api.authListAccounts();
      this.error = null;
    } catch (e) {
      this.error = errorMessage(e);
    } finally {
      this.loaded = true;
    }
  }

  /** Runs the whole device-code sign-in; resolves when it succeeds, fails or is cancelled. */
  async signIn(): Promise<void> {
    if (this.flow.phase !== 'idle') return;
    this.error = null;
    this.flow = { phase: 'starting' };
    try {
      const code = await api.authBeginDeviceFlow();
      this.flow = { phase: 'waiting', code, expiresAt: Date.now() + code.expires_in * 1000 };
      await api.authPoll();
      await this.load();
    } catch (e) {
      // Cancelling is a normal outcome, not an error worth showing.
      if (errorCode(e) !== 'auth_cancelled') this.error = errorMessage(e);
    } finally {
      this.flow = { phase: 'idle' };
    }
  }

  async cancel(): Promise<void> {
    try {
      await api.authCancel();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async openVerificationPage(): Promise<void> {
    try {
      await api.authOpenVerificationPage();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async setActive(id: string): Promise<void> {
    try {
      await api.authSetActive(id);
      await this.load();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async remove(id: string): Promise<void> {
    try {
      await api.authRemove(id);
      await this.load();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }
}

export const accounts = new AccountsStore();

// Typed IPC layer. Every call to the Rust backend goes through here.
import { invoke } from '@tauri-apps/api/core';

export type Theme = 'system' | 'dark' | 'light';

export interface Settings {
  theme: Theme;
  accent_color: string;
  animated_background: boolean;
  reduce_motion: boolean;
  max_concurrent_downloads: number;
  default_ram_mb: number | null;
  check_updates_on_start: boolean;
  log_level: 'error' | 'warn' | 'info' | 'debug' | 'trace';
}

export interface AppInfo {
  name: string;
  version: string;
  os: string;
  arch: string;
  portable: boolean;
  data_dir: string;
}

export interface AppError {
  code: string;
  message: string;
}

export interface Account {
  id: string;
  username: string;
  skin_url: string | null;
  skin_variant: 'classic' | 'slim' | null;
  cape_url: string | null;
  last_used: number | null;
  is_active: boolean;
  /** The Microsoft sign-in expired or was revoked: sign in again. */
  needs_reauth: boolean;
}

export interface DeviceCodeInfo {
  user_code: string;
  verification_uri: string;
  expires_in: number;
}

const inTauri = typeof window !== 'undefined' && '__TAURI_INTERNALS__' in window;

async function call<T>(cmd: string, args?: Record<string, unknown>): Promise<T> {
  if (inTauri) return invoke<T>(cmd, args);
  if (import.meta.env.DEV) {
    // Plain `vite dev` in a browser: use the in-memory stand-in. Stripped from production builds.
    const { mockInvoke } = await import('./mock');
    return mockInvoke(cmd, args) as Promise<T>;
  }
  throw { code: 'no_backend', message: 'Kiln must be opened through its desktop app.' } satisfies AppError;
}

/** Plain-English text for any error thrown by an IPC call. */
export function errorMessage(e: unknown): string {
  if (typeof e === 'object' && e !== null && 'message' in e && typeof e.message === 'string') {
    return e.message;
  }
  return 'Something went wrong. Try again, and check the logs if it keeps happening.';
}

export function errorCode(e: unknown): string | null {
  return typeof e === 'object' && e !== null && 'code' in e && typeof e.code === 'string' ? e.code : null;
}

export const api = {
  getAppInfo: () => call<AppInfo>('get_app_info'),
  getSettings: () => call<Settings>('get_settings'),
  setSetting: <K extends keyof Settings>(key: K, value: Settings[K]) =>
    call<Settings>('set_setting', { key, value }),

  // Accounts. No call here ever returns a token.
  authBeginDeviceFlow: () => call<DeviceCodeInfo>('auth_begin_device_flow'),
  /** Resolves when the user finishes signing in; rejects with code `auth_cancelled` if cancelled. */
  authPoll: () => call<Account>('auth_poll'),
  authCancel: () => call<void>('auth_cancel'),
  authOpenVerificationPage: () => call<void>('auth_open_verification_page'),
  authListAccounts: () => call<Account[]>('auth_list_accounts'),
  authSetActive: (id: string) => call<Account>('auth_set_active', { id }),
  authRemove: (id: string) => call<void>('auth_remove', { id }),
};

// Browser-only stand-in for the Rust backend, used by `npm run dev` outside Tauri.
import type { Account, AppError, AppInfo, Settings } from './api';

let settings: Settings = {
  theme: 'system',
  accent_color: '#5fb99a',
  animated_background: false,
  reduce_motion: false,
  max_concurrent_downloads: 8,
  default_ram_mb: null,
  check_updates_on_start: true,
  log_level: 'info',
};

let accounts: Account[] = [];
let cancelPoll: (() => void) | null = null;

const info: AppInfo = {
  name: 'Kiln',
  version: '0.1.0-dev',
  os: 'browser',
  arch: 'n/a',
  portable: false,
  data_dir: '(not available in the browser preview)',
};

export async function mockInvoke(cmd: string, args?: Record<string, unknown>): Promise<unknown> {
  switch (cmd) {
    case 'get_app_info':
      return info;
    case 'get_settings':
      return settings;
    case 'set_setting': {
      const key = String(args?.key);
      if (!(key in settings)) {
        throw { code: 'unknown_setting', message: `Kiln has no setting called "${key}".` } satisfies AppError;
      }
      settings = { ...settings, [key]: args?.value };
      return settings;
    }
    case 'auth_begin_device_flow':
      return { user_code: 'DEMO1234', verification_uri: 'https://www.microsoft.com/link', expires_in: 900 };
    case 'auth_open_verification_page':
      return undefined;
    case 'auth_poll':
      return new Promise((resolve, reject) => {
        const timer = setTimeout(() => {
          cancelPoll = null;
          const n = accounts.length + 1;
          const account: Account = {
            id: String(n).repeat(32),
            username: `Player_${n}`,
            skin_url: null,
            skin_variant: null,
            cape_url: null,
            last_used: Date.now() / 1000,
            is_active: true,
            needs_reauth: false,
          };
          accounts = [account, ...accounts.map((a) => ({ ...a, is_active: false }))];
          resolve(account);
        }, 4000);
        cancelPoll = () => {
          clearTimeout(timer);
          reject({ code: 'auth_cancelled', message: 'Sign-in was cancelled.' } satisfies AppError);
        };
      });
    case 'auth_cancel':
      cancelPoll?.();
      cancelPoll = null;
      return undefined;
    case 'auth_list_accounts':
      return accounts;
    case 'auth_set_active':
      accounts = accounts.map((a) => ({ ...a, is_active: a.id === args?.id }));
      return accounts.find((a) => a.is_active);
    case 'auth_remove':
      accounts = accounts.filter((a) => a.id !== args?.id);
      return undefined;
    default:
      throw { code: 'unknown_command', message: `Unknown command ${cmd}` } satisfies AppError;
  }
}

import { api, errorMessage, type AppInfo, type Settings } from './api';
import { applyTheme } from './theme';

class SettingsStore {
  value = $state<Settings | null>(null);
  info = $state<AppInfo | null>(null);
  error = $state<string | null>(null);

  async init(): Promise<void> {
    try {
      const [settings, info] = await Promise.all([api.getSettings(), api.getAppInfo()]);
      this.value = settings;
      this.info = info;
      this.error = null;
      applyTheme(settings);
      // Follow the OS while the theme is "system".
      window
        .matchMedia('(prefers-color-scheme: dark)')
        .addEventListener('change', () => this.value && applyTheme(this.value));
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async set<K extends keyof Settings>(key: K, value: Settings[K]): Promise<void> {
    try {
      this.value = await api.setSetting(key, value);
      this.error = null;
      applyTheme(this.value);
    } catch (e) {
      this.error = errorMessage(e);
    }
  }
}

import type { Settings } from './api';

function channel(v: number): number {
  const c = v / 255;
  return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
}

/** WCAG relative luminance of a `#rrggbb` color. */
export function luminance(hex: string): number {
  const n = Number.parseInt(hex.slice(1), 16);
  return 0.2126 * channel((n >> 16) & 255) + 0.7152 * channel((n >> 8) & 255) + 0.0722 * channel(n & 255);
}

/** Text color that stays readable on top of an accent-colored fill. */
export function onAccent(hex: string): string {
  return luminance(hex) > 0.3 ? '#10201b' : '#ffffff';
}

export function resolveTheme(theme: Settings['theme'], prefersDark: boolean): 'dark' | 'light' {
  return theme === 'system' ? (prefersDark ? 'dark' : 'light') : theme;
}

export function applyTheme(s: Settings): void {
  const root = document.documentElement;
  const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
  root.dataset.theme = resolveTheme(s.theme, prefersDark);
  root.style.setProperty('--accent', s.accent_color);
  root.style.setProperty('--on-accent', onAccent(s.accent_color));
  root.dataset.motion = s.reduce_motion ? 'reduced' : 'full';
  root.dataset.bg = s.animated_background ? 'animated' : 'plain';
}
