import { api, errorCode, errorMessage, type Account, type DeviceCodeInfo } from './api';

export type Flow =
  | { phase: 'idle' }
  | { phase: 'starting' }
  | { phase: 'waiting'; code: DeviceCodeInfo; expiresAt: number };

class AccountsStore {
  list = $state<Account[]>([]);
  loaded = $state(false);
  flow = $state<Flow>({ phase: 'idle' });
  error = $state<string | null>(null);

  async load(): Promise<void> {
    try {
      this.list = await api.authListAccounts();
      this.error = null;
    } catch (e) {
      this.error = errorMessage(e);
    } finally {
      this.loaded = true;
    }
  }

  /** Runs the whole device-code sign-in; resolves when it succeeds, fails or is cancelled. */
  async signIn(): Promise<void> {
    if (this.flow.phase !== 'idle') return;
    this.error = null;
    this.flow = { phase: 'starting' };
    try {
      const code = await api.authBeginDeviceFlow();
      this.flow = { phase: 'waiting', code, expiresAt: Date.now() + code.expires_in * 1000 };
      await api.authPoll();
      await this.load();
    } catch (e) {
      // Cancelling is a normal outcome, not an error worth showing.
      if (errorCode(e) !== 'auth_cancelled') this.error = errorMessage(e);
    } finally {
      this.flow = { phase: 'idle' };
    }
  }

  async cancel(): Promise<void> {
    try {
      await api.authCancel();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async openVerificationPage(): Promise<void> {
    try {
      await api.authOpenVerificationPage();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async setActive(id: string): Promise<void> {
    try {
      await api.authSetActive(id);
      await this.load();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async remove(id: string): Promise<void> {
    try {
      await api.authRemove(id);
      await this.load();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }
}

export const accounts = new AccountsStore();

const inTauri = typeof window !== 'undefined' && '__TAURI_INTERNALS__' in window;

async function call<T>(cmd: string, args?: Record<string, unknown>): Promise<T> {
  if (inTauri) return invoke<T>(cmd, args);
  if (import.meta.env.DEV) {
    // Plain `vite dev` in a browser: use the in-memory stand-in. Stripped from production builds.
    const { mockInvoke } = await import('./mock');
    return mockInvoke(cmd, args) as Promise<T>;
  }
  throw { code: 'no_backend', message: 'Kiln must be opened through its desktop app.' } satisfies AppError;
}

/** Plain-English text for any error thrown by an IPC call. */
export function errorMessage(e: unknown): string {
  if (typeof e === 'object' && e !== null && 'message' in e && typeof e.message === 'string') {
    return e.message;
  }
  return 'Something went wrong. Try again, and check the logs if it keeps happening.';
}

export function errorCode(e: unknown): string | null {
  return typeof e === 'object' && e !== null && 'code' in e && typeof e.code === 'string' ? e.code : null;
}

import type { Progress } from './api';

/** `1536` -> `1.5 KB`. */
export function formatBytes(n: number): string {
  if (!Number.isFinite(n) || n < 0) return '0 B';
  const units = ['B', 'KB', 'MB', 'GB', 'TB'];
  let value = n;
  let i = 0;
  while (value >= 1024 && i < units.length - 1) {
    value /= 1024;
    i++;
  }
  return `${i === 0 || value >= 100 ? Math.round(value) : value.toFixed(1)} ${units[i]}`;
}

/** Whole-number percentage of bytes done, or `null` when the total isn't known yet. */
export function progressPercent(p: Progress | null): number | null {
  if (!p || p.bytes_total <= 0) return null;
  return Math.min(100, Math.floor((p.bytes_done / p.bytes_total) * 100));
}

/** `7500` -> `2 h 5 min`; under a minute -> `less than a minute`. */
export function formatPlaytime(seconds: number): string {
  if (!Number.isFinite(seconds) || seconds < 60) return 'less than a minute';
  const minutes = Math.floor(seconds / 60);
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  return h === 0 ? `${m} min` : m === 0 ? `${h} h` : `${h} h ${m} min`;
}

class InstancesStore {
  list = $state<Instance[]>([]);
  imports = $state<ImportFile[]>([]);
  loaded = $state(false);
  error = $state<string | null>(null);
  /** A short confirmation, e.g. where an export was written. */
  notice = $state<string | null>(null);
  /** What is in progress: an instance id, or `import`. */
  busy = $state<string | null>(null);
  /** Download progress of an import in flight. */
  progress = $state<Progress | null>(null);
  private listening = false;

  /** Follow import progress. Safe to call more than once. */
  async listen(): Promise<void> {
    if (this.listening) return;
    this.listening = true;
    try {
      await listenProgress((p) => {
        if (p.task === 'import' && this.busy === 'import') this.progress = p.progress;
      });
    } catch {
      this.listening = false;
    }
  }

  private async run<T>(busy: string | null, work: () => Promise<T>): Promise<T | undefined> {
    this.error = null;
    this.notice = null;
    this.busy = busy;
    this.progress = null;
    try {
      return await work();
    } catch (e) {
      this.error = errorMessage(e);
      return undefined;
    } finally {
      this.busy = null;
      this.progress = null;
    }
  }

  async load(): Promise<void> {
    try {
      this.list = await api.instanceList();
      this.error = null;
    } catch (e) {
      this.error = errorMessage(e);
    } finally {
      this.loaded = true;
    }
  }

  async loadImports(): Promise<void> {
    try {
      this.imports = await api.instanceListImports();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async create(name: string, mc: string, loader: Instance['loader'], loaderVersion: string | null): Promise<boolean> {
    const made = await this.run(null, () => api.instanceCreate(name, mc, loader, loaderVersion));
    if (made) await this.load();
    return made !== undefined;
  }

  async rename(id: string, name: string): Promise<boolean> {
    const ok = await this.run(id, () => api.instanceRename(id, name));
    if (ok) await this.load();
    return ok !== undefined;
  }

  async duplicate(id: string, name: string): Promise<void> {
    if ((await this.run(id, () => api.instanceClone(id, name))) !== undefined) await this.load();
  }

  async remove(id: string): Promise<void> {
    const done = await this.run(id, async () => {
      await api.instanceDelete(id);
      return true;
    });
    if (done) await this.load();
  }

  async saveSettings(id: string, ramMb: number | null, jvmArgs: string | null): Promise<boolean> {
    const ok = await this.run(id, () => api.instanceUpdateSettings(id, ramMb, jvmArgs));
    if (ok) await this.load();
    return ok !== undefined;
  }

  async export(id: string, format: 'kiln' | 'mrpack'): Promise<void> {
    const out = await this.run(id, () => api.instanceExport(id, format));
    if (!out) return;
    let notice = `Saved ${out.file} in Kiln's exports folder.`;
    if (out.listed > 0) notice += ` ${out.listed} mods are listed by their Modrinth hash instead of being copied in.`;
    if (out.bundled_unverified.length > 0) {
      const names = out.bundled_unverified.slice(0, 5).join(', ');
      const more = out.bundled_unverified.length > 5 ? ` and ${out.bundled_unverified.length - 5} more` : '';
      notice += ` Modrinth doesn't know ${names}${more}, so they were bundled into the pack. Check you're allowed to share them.`;
    }
    this.notice = notice;
  }

  async importFile(name: string): Promise<void> {
    const made = await this.run('import', () => api.instanceImport(name));
    if (made) {
      this.notice = `Added ${made.name} to your instances.`;
      await Promise.all([this.load(), this.loadImports()]);
    }
  }

  async openFolder(which: 'imports' | 'exports' | 'instance', id?: string): Promise<void> {
    await this.run(null, () => api.openFolder(which, id));
  }
}

import { api, errorMessage, listenLaunch, type LaunchStatus, type LogLine } from './api';

/** Console lines kept in memory; the backend keeps the same number for late openers. */
const MAX_LINES = 2000;

class LaunchStore {
  status = $state<LaunchStatus>({ state: 'idle' });
  lines = $state<LogLine[]>([]);
  error = $state<string | null>(null);
  /** Set when live progress and output can't be received; separate so it isn't overwritten. */
  eventsError = $state<string | null>(null);
  private started = false;

  readonly active = $derived(this.status.state === 'preparing' || this.status.state === 'running');

  /** Subscribe to backend events once, then catch up on anything already happened. */
  async init(): Promise<void> {
    if (this.started) return;
    this.started = true;
    try {
      await listenLaunch(
        (s) => (this.status = s),
        (l) => this.push(l),
      );
      this.status = await api.launchStatus();
      this.lines = await api.launchLog();
    } catch (e) {
      this.started = false;
      this.eventsError = errorMessage(e);
    }
  }

  private push(line: LogLine) {
    this.lines.push(line);
    if (this.lines.length > MAX_LINES) this.lines.splice(0, this.lines.length - MAX_LINES);
  }

  async start(instanceId: string): Promise<void> {
    if (this.active) return;
    this.error = null;
    try {
      await api.launchStart(instanceId);
    } catch (e) {
      this.error = errorMessage(e);
    }
  }

  async stop(): Promise<void> {
    try {
      await api.launchStop();
    } catch (e) {
      this.error = errorMessage(e);
    }
  }
}

import type { InstallPlan, ModEntry, RollbackReport, UpdateInfo } from './api';

/** `1234` -> `1.2K`, `2500000` -> `2.5M`. */
export function formatCount(n: number | null | undefined): string {
  if (n === null || n === undefined || !Number.isFinite(n) || n < 0) return '–';
  if (n < 1000) return String(Math.round(n));
  const units: [number, string][] = [
    [1e9, 'B'],
    [1e6, 'M'],
    [1e3, 'K'],
  ];
  for (const [size, suffix] of units) {
    if (n >= size) {
      const v = n / size;
      return `${v >= 100 ? Math.round(v) : v.toFixed(1).replace(/\.0$/, '')}${suffix}`;
    }
  }
  return String(n);
}

/** One line about where a mod came from. Unverified jars get no author, downloads or licence. */
export function sourceLine(m: ModEntry): string {
  if (m.source !== 'modrinth') {
    return 'Not from Modrinth: Kiln can’t verify who made this file or check it for updates.';
  }
  const parts = [m.author ? `by ${m.author}` : null, m.downloads !== null ? `${formatCount(m.downloads)} downloads` : null, m.license]
    .filter((p): p is string => !!p);
  return parts.join(' · ') || 'From Modrinth';
}

/** What an install will do, in one sentence, for the confirmation panel. */
export function planSummary(plan: InstallPlan): string {
  const wanted = plan.mods.filter((m) => m.reason === 'requested').length;
  const deps = plan.mods.length - wanted;
  if (plan.mods.length === 0) return 'Everything needed is already installed.';
  const base = wanted === 1 ? 'Installs 1 mod' : `Installs ${wanted} mods`;
  return deps > 0 ? `${base} and ${deps} ${deps === 1 ? 'dependency' : 'dependencies'}.` : `${base}.`;
}

export function updateLabel(u: UpdateInfo): string {
  const from = u.current_version ? `${u.current_version} → ` : '';
  return `${from}${u.new_version}${u.channel === 'release' ? '' : ` (${u.channel})`}`;
}

export function rollbackSummary(r: RollbackReport): string {
  const parts: string[] = [];
  if (r.restored.length > 0) parts.push(`put back ${r.restored.length === 1 ? r.restored[0] : `${r.restored.length} mods`}`);
  if (r.removed.length > 0) parts.push(`removed ${r.removed.length === 1 ? r.removed[0] : `${r.removed.length} added mods`}`);
  const done = parts.length > 0 ? `Kiln ${parts.join(' and ')}.` : 'Nothing could be undone.';
  return r.failed.length > 0 ? `${done} Couldn’t restore: ${r.failed.join(', ')}.` : done;
}

import {
  api,
  errorMessage,
  listenProgress,
  listenRollback,
  type InstallPlan,
  type Instance,
  type ModEntry,
  type ProjectKind,
  type Progress,
  type SearchHit,
  type SortOrder,
  type UpdateInfo,
} from './api';
import { rollbackSummary } from './mods-format';

/** Mods, modpacks and their updates for one instance at a time. */
class ModsStore {
  instanceId = $state<string | null>(null);
  installed = $state<ModEntry[]>([]);
  updates = $state<UpdateInfo[]>([]);
  checkedUpdates = $state(false);
  loading = $state(false);

  hits = $state<SearchHit[]>([]);
  total = $state(0);
  searched = $state(false);
  searching = $state(false);

  /** The install plan being reviewed, for `planFor`. */
  plan = $state<InstallPlan | null>(null);
  planFor = $state<{ id: string; title: string } | null>(null);

  /** The operation in flight: `plan`, `install`, `update:<id>`, `update-all`, `rollback`, `modpack`... */
  busy = $state<string | null>(null);
  progress = $state<Progress | null>(null);
  error = $state<string | null>(null);
  notice = $state<string | null>(null);
  private started = false;

  /** Listen for progress and automatic rollbacks, once. */
  async init(): Promise<void> {
    if (this.started) return;
    this.started = true;
    try {
      await listenProgress((p) => {
        if (this.busy !== null) this.progress = p.progress;
      });
      await listenRollback((r) => {
        this.notice = `The game didn’t start after a mod update. ${rollbackSummary(r)}`;
        if (r.instance_id === this.instanceId) void this.load();
      });
    } catch (e) {
      this.started = false;
      this.error = errorMessage(e);
    }
  }

  private async run<T>(busy: string, work: () => Promise<T>): Promise<T | undefined> {
    this.error = null;
    this.notice = null;
    this.busy = busy;
    this.progress = null;
    try {
      return await work();
    } catch (e) {
      this.error = errorMessage(e);
      return undefined;
    } finally {
      this.busy = null;
      this.progress = null;
    }
  }

  async select(instanceId: string | null): Promise<void> {
    this.instanceId = instanceId;
    this.installed = [];
    this.updates = [];
    this.checkedUpdates = false;
    this.plan = null;
    this.planFor = null;
    this.error = null;
    this.notice = null;
    if (instanceId) await this.load();
  }

  async load(): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    this.loading = true;
    try {
      const list = await api.modsList(id);
      if (this.instanceId === id) this.installed = list;
    } catch (e) {
      this.error = errorMessage(e);
    } finally {
      this.loading = false;
    }
  }

  async search(query: string, kind: ProjectKind, sort: SortOrder, mc: string | null, loader: Instance['loader'] | null): Promise<void> {
    this.searching = true;
    this.error = null;
    try {
      const page = await api.modSearch({
        query,
        kind,
        sort,
        mc_version: mc,
        loader: kind === 'mod' ? loader : null,
      });
      this.hits = page.hits;
      this.total = page.total;
      this.searched = true;
    } catch (e) {
      this.error = errorMessage(e);
    } finally {
      this.searching = false;
    }
  }

  async review(project: { id: string; title: string }): Promise<void> {
    if (!this.instanceId) return;
    const id = this.instanceId;
    const plan = await this.run('plan', () => api.modInstallPlan(id, project.id));
    if (plan) {
      this.plan = plan;
      this.planFor = project;
    }
  }

  cancelReview(): void {
    this.plan = null;
    this.planFor = null;
  }

  async confirmInstall(optional: string[]): Promise<void> {
    const id = this.instanceId;
    const target = this.planFor;
    if (!id || !target) return;
    const done = await this.run('install', () => api.modInstall(id, target.id, optional));
    if (done) {
      this.notice = `Installed ${target.title}${done.mods.length > 1 ? ` and ${done.mods.length - 1} more` : ''}.`;
      this.cancelReview();
      this.updates = [];
      this.checkedUpdates = false;
      await this.load();
    }
  }

  async checkUpdates(): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    const found = await this.run('check', () => api.modsCheckUpdates(id));
    if (found) {
      this.updates = found;
      this.checkedUpdates = true;
      await this.load();
    }
  }

  updateFor(modId: number): UpdateInfo | undefined {
    return this.updates.find((u) => u.mod_id === modId);
  }

  async update(modId: number): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    const ok = await this.run(`update:${modId}`, () => api.modUpdate(id, modId));
    if (ok !== undefined) {
      this.notice = ok ? 'Updated. If the game doesn’t start afterwards, Kiln will undo the update.' : 'Already up to date.';
      await this.afterChange();
    }
  }

  async updateAll(): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    const report = await this.run('update-all', () => api.modsUpdateAll(id));
    if (report) {
      const bits = [`Updated ${report.updated.length}.`];
      if (report.skipped_pinned > 0) bits.push(`${report.skipped_pinned} pinned, left alone.`);
      if (report.failed.length > 0) bits.push(`${report.failed.length} failed: ${report.failed.map((f) => f.title).join(', ')}.`);
      this.notice = bits.join(' ');
      await this.afterChange();
    }
  }

  private async afterChange(): Promise<void> {
    await this.load();
    const id = this.instanceId;
    if (!id) return;
    try {
      this.updates = await api.modsCheckUpdates(id);
    } catch {
      this.updates = [];
    }
  }

  async setPinned(modId: number, pinned: boolean): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    if ((await this.run(`pin:${modId}`, async () => (await api.modSetPinned(id, modId, pinned), true))) !== undefined) {
      this.updates = this.updates.map((u) => (u.mod_id === modId ? { ...u, pinned } : u));
      await this.load();
    }
  }

  async setEnabled(modId: number, enabled: boolean): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    if ((await this.run(`toggle:${modId}`, async () => (await api.modSetEnabled(id, modId, enabled), true))) !== undefined) await this.load();
  }

  async remove(modId: number): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    if ((await this.run(`remove:${modId}`, async () => (await api.modRemove(id, modId), true))) !== undefined) {
      this.updates = this.updates.filter((u) => u.mod_id !== modId);
      await this.load();
    }
  }

  async rollback(): Promise<void> {
    const id = this.instanceId;
    if (!id) return;
    const report = await this.run('rollback', () => api.modsRollback(id));
    if (report) {
      this.notice = rollbackSummary(report);
      await this.afterChange();
    }
  }

  async installModpack(projectId: string, title: string): Promise<Instance | undefined> {
    const made = await this.run('modpack', () => api.modpackInstall(projectId));
    if (made) this.notice = `Added ${made.name} to your instances (from ${title}).`;
    return made;
  }
}

export const mods = new ModsStore();

import type { Settings } from './api';

function channel(v: number): number {
  const c = v / 255;
  return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
}

/** WCAG relative luminance of a `#rrggbb` color. */
export function luminance(hex: string): number {
  const n = Number.parseInt(hex.slice(1), 16);
  return 0.2126 * channel((n >> 16) & 255) + 0.7152 * channel((n >> 8) & 255) + 0.0722 * channel(n & 255);
}

/** Text color that stays readable on top of an accent-colored fill. */
export function onAccent(hex: string): string {
  return luminance(hex) > 0.3 ? '#10201b' : '#ffffff';
}

export function resolveTheme(theme: Settings['theme'], prefersDark: boolean): 'dark' | 'light' {
  return theme === 'system' ? (prefersDark ? 'dark' : 'light') : theme;
}

export function applyTheme(s: Settings): void {
  const root = document.documentElement;
  const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
  root.dataset.theme = resolveTheme(s.theme, prefersDark);
  root.style.setProperty('--accent', s.accent_color);
  root.style.setProperty('--on-accent', onAccent(s.accent_color));
  root.dataset.motion = s.reduce_motion ? 'reduced' : 'full';
  root.dataset.bg = s.animated_background ? 'animated' : 'plain';
}

    void mods.search(
      query.trim(),
      kind,
      sort,
      fitInstance && current ? current.mc_version : null,
      fitInstance && current ? current.loader : null,
    );
  }

  function openBrowse() {
    tab = 'browse';
    if (!mods.searched) runSearch();
  }

  async function review(hit: { id: string; title: string }) {
    chosenOptional = [];
    await mods.review(hit);
  }

  function toggleOptional(id: string, on: boolean) {
    chosenOptional = on ? [...chosenOptional, id] : chosenOptional.filter((x) => x !== id);
  }

  async function installPack(id: string, title: string) {
    confirmPack = null;
    const made = await mods.installModpack(id, title);
    if (made) {
      await instances.load();
      router.go('instances');
    }
  }

            {#if hit.kind === 'modpack'}
              {#if confirmPack === hit.id}
                <div class="flex flex-col items-end gap-1 text-sm">
                  <span class="text-muted">Creates a new instance.</span>
                  <span class="flex gap-3">
                    <button type="button" class="cursor-pointer font-medium text-accent hover:underline" onclick={() => installPack(hit.id, hit.title)}>Install</button>
                    <button type="button" class="cursor-pointer text-muted underline" onclick={() => (confirmPack = null)}>Cancel</button>
                  </span>
                </div>
              {:else}
                <button
                  type="button"
                  class="{button} border border-line hover:bg-raised"
                  disabled={mods.busy !== null}
                  onclick={() => (confirmPack = hit.id)}>{mods.busy === 'modpack' ? 'Installing…' : 'Install pack'}</button
                >
              {/if}
            {:else if installedProjects.has(hit.id)}
              <span class="self-center text-sm text-muted">Installed</span>
            {:else}
              <button
                type="button"
                class="{button} border border-line hover:bg-raised"
                disabled={mods.busy !== null}
                onclick={() => review(hit)}>Install</button
              >
            {/if}
          </li>
        {/each}
      </ul>
      {#if mods.total > mods.hits.length}
        <p class="mt-3 text-sm text-muted">Showing {mods.hits.length} of {formatCount(mods.total)}. Narrow the search to see others.</p>
      {/if}
    {/if}
  {/if}
</section>
