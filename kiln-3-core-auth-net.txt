//! Non-secret account metadata in SQLite. Tokens never touch this table.

use serde::Serialize;
use sqlx::FromRow;

use super::error::AuthError;
use super::minecraft::Profile;
use crate::db::Db;
use crate::Result;

pub const TEXTURE_BASE: &str = "https://textures.minecraft.net/texture/";

pub fn texture_url(hash: &str) -> String {
    format!("{TEXTURE_BASE}{hash}")
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
#[serde(rename_all = "lowercase")]
pub enum SkinVariant {
    Classic,
    Slim,
}

impl SkinVariant {
    pub fn parse(s: &str) -> Option<Self> {
        match s.to_ascii_lowercase().as_str() {
            "classic" => Some(Self::Classic),
            "slim" => Some(Self::Slim),
            _ => None,
        }
    }

    fn as_str(self) -> &'static str {
        match self {
            Self::Classic => "classic",
            Self::Slim => "slim",
        }
    }
}

/// Accounts are identified by the Minecraft player UUID: 32 lowercase hex characters.
pub fn is_valid_account_id(id: &str) -> bool {
    id.len() == 32 && id.bytes().all(|b| matches!(b, b'0'..=b'9' | b'a'..=b'f'))
}

pub fn validate_account_id(id: &str) -> Result<()> {
    if is_valid_account_id(id) {
        Ok(())
    } else {
        Err(AuthError::InvalidAccountId.into())
    }
}

#[derive(Debug, Clone, Serialize, PartialEq, Eq)]
pub struct Account {
    pub id: String,
    pub username: String,
    pub skin_url: Option<String>,
    pub skin_variant: Option<SkinVariant>,
    pub cape_url: Option<String>,
    pub last_used: Option<i64>,
    pub is_active: bool,
    /// The Microsoft sign-in was revoked or expired: the account must be signed in again.
    pub needs_reauth: bool,
}

#[derive(FromRow)]
struct Row {
    id: String,
    username: String,
    skin_hash: Option<String>,
    skin_variant: Option<String>,
    cape_hash: Option<String>,
    last_used: Option<i64>,
    is_active: i64,
    needs_reauth: i64,
}

impl From<Row> for Account {
    fn from(r: Row) -> Self {
        Self {
            id: r.id,
            username: r.username,
            skin_url: r.skin_hash.as_deref().map(texture_url),
            skin_variant: r.skin_variant.as_deref().and_then(SkinVariant::parse),
            cape_url: r.cape_hash.as_deref().map(texture_url),
            last_used: r.last_used,
            is_active: r.is_active != 0,
            needs_reauth: r.needs_reauth != 0,
        }
    }
}

const COLUMNS: &str =
    "id, username, skin_hash, skin_variant, cape_hash, last_used, is_active, needs_reauth";

pub async fn list(db: &Db) -> Result<Vec<Account>> {
    let rows: Vec<Row> = sqlx::query_as(&format!(
        "SELECT {COLUMNS} FROM accounts ORDER BY is_active DESC, last_used DESC, username COLLATE NOCASE"
    ))
    .fetch_all(db)
    .await?;
    Ok(rows.into_iter().map(Account::from).collect())
}

pub async fn get(db: &Db, id: &str) -> Result<Option<Account>> {
    let row: Option<Row> = sqlx::query_as(&format!("SELECT {COLUMNS} FROM accounts WHERE id = ?1"))
        .bind(id)
        .fetch_optional(db)
        .await?;
    Ok(row.map(Account::from))
}

/// Insert or update a freshly signed-in account and make it the active one.
pub async fn upsert_signed_in(db: &Db, p: &Profile, now: i64) -> Result<Account> {
    let mut tx = db.begin().await?;
    sqlx::query("UPDATE accounts SET is_active = 0 WHERE is_active = 1")
        .execute(&mut *tx)
        .await?;
    sqlx::query(
        "INSERT INTO accounts (id, username, skin_hash, skin_variant, cape_hash, last_used, is_active, needs_reauth) \
         VALUES (?1, ?2, ?3, ?4, ?5, ?6, 1, 0) \
         ON CONFLICT (id) DO UPDATE SET username = excluded.username, skin_hash = excluded.skin_hash, \
           skin_variant = excluded.skin_variant, cape_hash = excluded.cape_hash, \
           last_used = excluded.last_used, is_active = 1, needs_reauth = 0",
    )
    .bind(&p.id)
    .bind(&p.name)
    .bind(&p.skin_hash)
    .bind(p.skin_variant.map(SkinVariant::as_str))
    .bind(&p.cape_hash)
    .bind(now)
    .execute(&mut *tx)
    .await?;
    tx.commit().await?;
    get(db, &p.id)
        .await?
        .ok_or_else(|| AuthError::AccountNotFound.into())
}

/// Refresh profile details after a token refresh (the player may have renamed or changed
/// skin). Keeps the active flag; clears `needs_reauth`.
pub async fn update_profile(db: &Db, p: &Profile) -> Result<()> {
    sqlx::query(
        "UPDATE accounts SET username = ?2, skin_hash = ?3, skin_variant = ?4, cape_hash = ?5, needs_reauth = 0 \
         WHERE id = ?1",
    )
    .bind(&p.id)
    .bind(&p.name)
    .bind(&p.skin_hash)
    .bind(p.skin_variant.map(SkinVariant::as_str))
    .bind(&p.cape_hash)
    .execute(db)
    .await?;
    Ok(())
}

pub async fn mark_needs_reauth(db: &Db, id: &str) -> Result<()> {
    sqlx::query("UPDATE accounts SET needs_reauth = 1 WHERE id = ?1")
        .bind(id)
        .execute(db)
        .await?;
    Ok(())
}

pub async fn set_active(db: &Db, id: &str, now: i64) -> Result<Account> {
    let mut tx = db.begin().await?;
    let exists: Option<(i64,)> = sqlx::query_as("SELECT 1 FROM accounts WHERE id = ?1")
        .bind(id)
        .fetch_optional(&mut *tx)
        .await?;
    if exists.is_none() {
        return Err(AuthError::AccountNotFound.into());
    }
    sqlx::query("UPDATE accounts SET is_active = 0 WHERE is_active = 1")
        .execute(&mut *tx)
        .await?;
    sqlx::query("UPDATE accounts SET is_active = 1, last_used = ?2 WHERE id = ?1")
        .bind(id)
        .bind(now)
        .execute(&mut *tx)
        .await?;
    tx.commit().await?;
    get(db, id)
        .await?
        .ok_or_else(|| AuthError::AccountNotFound.into())
}

/// Delete an account row. If it was the active one, the most recently used remaining
/// account becomes active.
pub async fn remove(db: &Db, id: &str) -> Result<()> {
    let mut tx = db.begin().await?;
    let was_active: Option<(i64,)> = sqlx::query_as("SELECT is_active FROM accounts WHERE id = ?1")
        .bind(id)
        .fetch_optional(&mut *tx)
        .await?;
    let Some((was_active,)) = was_active else {
        return Err(AuthError::AccountNotFound.into());
    };
    sqlx::query("DELETE FROM accounts WHERE id = ?1")
        .bind(id)
        .execute(&mut *tx)
        .await?;
    if was_active != 0 {
        sqlx::query(
            "UPDATE accounts SET is_active = 1 WHERE id = \
             (SELECT id FROM accounts ORDER BY last_used DESC, username COLLATE NOCASE LIMIT 1)",
        )
        .execute(&mut *tx)
        .await?;
    }
    tx.commit().await?;
    Ok(())
}

        detail: String,

/// Build an "unexpected response" error. `detail` must never contain a token or body.
pub(crate) fn unexpected(service: &'static str, detail: impl Into<String>) -> Error {
    AuthError::Unexpected {
        service,
        detail: detail.into(),
    }
    .into()
}

//! Where refresh tokens live: the operating system's keychain, and nowhere else.
//! (Windows Credential Manager, macOS Keychain, Secret Service on Linux.)
//!
//! Secrets are stored as raw bytes. The string API on Windows stores UTF-16, which would
//! double a ~1.5 KB Microsoft refresh token and risk exceeding Credential Manager's
//! 2,560-byte limit.

use crate::{Error, Result};

pub trait SecretStore: Send + Sync {
    fn set(&self, account_id: &str, secret: &[u8]) -> Result<()>;
    fn get(&self, account_id: &str) -> Result<Option<Vec<u8>>>;
    /// Deleting something that doesn't exist is not an error.
    fn delete(&self, account_id: &str) -> Result<()>;
}

#[derive(Debug, Clone)]
pub struct KeyringStore {
    service: String,
}

impl KeyringStore {
    pub fn new(service: impl Into<String>) -> Self {
        Self {
            service: service.into(),
        }
    }

    fn entry(&self, account_id: &str) -> Result<keyring::Entry> {
        keyring::Entry::new(&self.service, &format!("msa-refresh-token:{account_id}"))
            .map_err(keychain_err)
    }
}

fn keychain_err(e: keyring::Error) -> Error {
    Error::Keychain(e.to_string())
}

impl SecretStore for KeyringStore {
    fn set(&self, account_id: &str, secret: &[u8]) -> Result<()> {
        self.entry(account_id)?
            .set_secret(secret)
            .map_err(keychain_err)
    }

    fn get(&self, account_id: &str) -> Result<Option<Vec<u8>>> {
        match self.entry(account_id)?.get_secret() {
            Ok(v) => Ok(Some(v)),
            Err(keyring::Error::NoEntry) => Ok(None),
            Err(e) => Err(keychain_err(e)),
        }
    }

    fn delete(&self, account_id: &str) -> Result<()> {
        match self.entry(account_id)?.delete_credential() {
            Ok(()) | Err(keyring::Error::NoEntry) => Ok(()),
            Err(e) => Err(keychain_err(e)),
        }
    }
}

/// In-memory store for tests. Deliberately not available outside `cfg(test)`, so a
/// production build can't accidentally keep tokens in an unprotected place.
#[cfg(test)]
#[derive(Default)]
pub struct MemoryStore(std::sync::Mutex<std::collections::HashMap<String, Vec<u8>>>);

#[cfg(test)]
impl MemoryStore {
    pub fn snapshot(&self) -> std::collections::HashMap<String, Vec<u8>> {
        self.0.lock().unwrap().clone()
    }
}

#[cfg(test)]
impl SecretStore for MemoryStore {
    fn set(&self, id: &str, secret: &[u8]) -> Result<()> {
        self.0.lock().unwrap().insert(id.into(), secret.to_vec());
        Ok(())
    }
    fn get(&self, id: &str) -> Result<Option<Vec<u8>>> {
        Ok(self.0.lock().unwrap().get(id).cloned())
    }
    fn delete(&self, id: &str) -> Result<()> {
        self.0.lock().unwrap().remove(id);
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn memory_store_semantics() {
        let s = MemoryStore::default();
        assert_eq!(s.get("a").unwrap(), None);
        s.set("a", b"one").unwrap();
        s.set("a", b"two").unwrap();
        assert_eq!(s.get("a").unwrap().as_deref(), Some(&b"two"[..]));
        s.delete("a").unwrap();
        s.delete("a").unwrap(); // idempotent
        assert_eq!(s.get("a").unwrap(), None);
    }

    /// Round trip against the real OS keychain. Ignored by default because CI runners and
    /// containers often have no unlocked keychain (on Linux it needs a running Secret
    /// Service such as gnome-keyring). Run with `cargo test -- --ignored keyring`.
    #[test]
    #[ignore = "needs an unlocked OS keychain"]
    fn keyring_round_trip_with_a_realistic_token_size() {
        let store = KeyringStore::new(format!("kiln-test-{}", std::process::id()));
        let id = "11111111111111112222222222222222";
        // About the size of a real Microsoft refresh token, and above the 1,280 characters
        // a UTF-16 encoding would allow on Windows.
        let token = "M.C507_BAY.0.U.".to_string() + &"x".repeat(1900);

        store.set(id, token.as_bytes()).unwrap();
        assert_eq!(store.get(id).unwrap().as_deref(), Some(token.as_bytes()));
        store.delete(id).unwrap();
        assert_eq!(store.get(id).unwrap(), None);
        store.delete(id).unwrap();
    }
}

#[derive(Deserialize)]
struct RawTexture {
    state: Option<String>,
    url: String,
    variant: Option<String>,

/// Extract the texture hash from a Mojang texture URL. Mojang serves these over plain
/// `http://`; only the host and a hex hash are trusted, and the UI rebuilds an HTTPS URL.
pub fn texture_hash(url: &str) -> Option<String> {
    let url = Url::parse(url).ok()?;
    if url.host_str() != Some("textures.minecraft.net") {
        return None;
    }
    let hash = url.path().strip_prefix("/texture/")?;
    let ok = (8..=64).contains(&hash.len())
        && hash.bytes().all(|b| matches!(b, b'0'..=b'9' | b'a'..=b'f'));
    ok.then(|| hash.to_owned())
}

fn is_active(t: &RawTexture) -> bool {
    t.state
        .as_deref()
        .is_some_and(|s| s.eq_ignore_ascii_case("ACTIVE"))
}

fn valid_name(name: &str) -> bool {
    (1..=16).contains(&name.len()) && name.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'_')
}

fn parse_profile(raw: RawProfile) -> Result<Profile> {
    let id = raw.id.to_ascii_lowercase().replace('-', "");
    if !is_valid_account_id(&id) || !valid_name(&raw.name) {
        return Err(unexpected(
            "Minecraft",
            "profile id or name failed validation",
        ));
    }
    let skin = raw
        .skins
        .iter()
        .find(|t| is_active(t))
        .or(raw.skins.first());
    let cape = raw.capes.iter().find(|t| is_active(t));
    Ok(Profile {
        id,
        name: raw.name,
        skin_hash: skin.and_then(|t| texture_hash(&t.url)),
        skin_variant: skin
            .and_then(|t| t.variant.as_deref())
            .and_then(SkinVariant::parse),
        cape_hash: cape.and_then(|t| texture_hash(&t.url)),
    })
}

pub async fn login_with_xbox(
    http: &HttpClient,
    ep: &Endpoints,
    xsts: &XboxToken,
) -> Result<McLogin> {
    let body = json!({ "identityToken": format!("XBL3.0 x={};{}", xsts.uhs, xsts.token.expose()) });
    let resp = http
        .execute(Method::POST, &ep.minecraft_login, Body::Json(&body), None)
        .await?;
    match resp.status {
        // Mojang answers 403 "Invalid app registration" for client ids it hasn't approved.
        403 => Err(AuthError::AppNotApproved.into()),
        s if !(200..300).contains(&s) => {
            Err(unexpected("Minecraft", format!("HTTP {s} from login")))
        }
        _ => {
            let body: LoginResponse = resp.json()?;
            Ok(McLogin {
                access_token: SecretString::new(body.access_token),
                expires_in: Duration::from_secs(body.expires_in.clamp(60, 172_800)),
            })
        }
    }
}

pub async fn fetch_profile(
    http: &HttpClient,
    ep: &Endpoints,
    token: &SecretString,
) -> Result<Profile> {
    let resp = http
        .execute(
            Method::GET,
            &ep.minecraft_profile,
            Body::Empty,

//! Microsoft identity platform, device-code flow.
//! <https://learn.microsoft.com/entra/identity-platform/v2-oauth2-device-code>

use std::time::Duration;

use reqwest::{Method, Url};
use serde::Deserialize;

use super::endpoints::Endpoints;
use super::error::{unexpected, AuthError};
use super::secret::SecretString;
use crate::net::{Body, HttpClient};
use crate::Result;

const SCOPE: &str = "XboxLive.signin offline_access";
const DEVICE_GRANT: &str = "urn:ietf:params:oauth:grant-type:device_code";
/// Where the server may send the user to enter the code. Checked because the UI offers
/// to open this address in the browser.
const VERIFICATION_HOSTS: &[&str] = &[
    "www.microsoft.com",
    "microsoft.com",
    "login.microsoftonline.com",
    "login.live.com",
    "aka.ms",
];

pub struct DeviceCode {
    pub device_code: SecretString,
    pub user_code: String,
    pub verification_uri: String,
    pub expires_in: Duration,
    pub interval: Duration,
}

pub struct MsTokens {
    pub access_token: SecretString,
    pub refresh_token: Option<SecretString>,
}

pub enum PollOutcome {
    Tokens(MsTokens),
    Pending,
    SlowDown,
}

#[derive(Deserialize)]
struct DeviceCodeResponse {
    device_code: String,
    user_code: String,
    verification_uri: String,
    expires_in: u64,
    interval: Option<u64>,
}

#[derive(Deserialize)]
struct TokenResponse {
    access_token: Option<String>,
    refresh_token: Option<String>,
    error: Option<String>,
}

fn short(code: &str) -> String {
    code.chars().take(64).collect()
}

fn valid_user_code(code: &str) -> bool {
    (1..=32).contains(&code.len()) && code.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'-')
}

fn valid_verification_uri(uri: &str) -> bool {
    Url::parse(uri).is_ok_and(|u| {
        u.scheme() == "https"
            && u.username().is_empty()
            && u.host_str()
                .is_some_and(|h| VERIFICATION_HOSTS.contains(&h))
    })
}

pub async fn request_device_code(
    http: &HttpClient,
    ep: &Endpoints,
    client_id: &str,
) -> Result<DeviceCode> {
    let resp = http
        .execute(
            Method::POST,
            &ep.device_code,
            Body::Form(&[("client_id", client_id), ("scope", SCOPE)]),
            None,
        )
        .await?;
    if !resp.is_success() {
        return Err(unexpected(
            "Microsoft",
            format!("HTTP {} requesting a sign-in code", resp.status),
        ));
    }
    let body: DeviceCodeResponse = resp.json()?;
    if !valid_user_code(&body.user_code) || !valid_verification_uri(&body.verification_uri) {
        return Err(unexpected(
            "Microsoft",
            "sign-in code or address failed validation",
        ));
    }
    Ok(DeviceCode {
        device_code: SecretString::new(body.device_code),
        user_code: body.user_code,
        verification_uri: body.verification_uri,
        expires_in: Duration::from_secs(body.expires_in.clamp(60, 1800)),
        interval: Duration::from_secs(body.interval.unwrap_or(5).clamp(1, 30)),
    })
}

pub async fn poll_token(
    http: &HttpClient,
    ep: &Endpoints,
    client_id: &str,
    device_code: &SecretString,
) -> Result<PollOutcome> {
    let resp = http
        .execute(
            Method::POST,
            &ep.token,
            Body::Form(&[
                ("grant_type", DEVICE_GRANT),
                ("client_id", client_id),
                ("device_code", device_code.expose()),
            ]),
            None,
        )
        .await?;
    let status = resp.status;
    let body: TokenResponse = resp.json().map_err(|_| {
        unexpected(
            "Microsoft",
            format!("HTTP {status} from the token endpoint"),
        )
    })?;

    if let Some(access) = body.access_token {
        return Ok(PollOutcome::Tokens(MsTokens {
            access_token: SecretString::new(access),
            refresh_token: body.refresh_token.map(SecretString::new),
        }));
    }
    match body.error.as_deref() {
        Some("authorization_pending") => Ok(PollOutcome::Pending),
        Some("slow_down") => Ok(PollOutcome::SlowDown),
        Some("authorization_declined") => Err(AuthError::FlowDeclined.into()),
        Some("expired_token" | "bad_verification_code") => Err(AuthError::FlowExpired.into()),
        other => Err(unexpected(
            "Microsoft",
            format!("error {}", short(other.unwrap_or("none"))),
        )),
    }
}

pub async fn refresh(
    http: &HttpClient,
    ep: &Endpoints,
    client_id: &str,
    refresh_token: &SecretString,
) -> Result<MsTokens> {
    let resp = http
        .execute(
            Method::POST,
            &ep.token,
            Body::Form(&[
                ("grant_type", "refresh_token"),
                ("client_id", client_id),
                ("refresh_token", refresh_token.expose()),
                ("scope", SCOPE),
            ]),
            None,
        )
        .await?;
    let status = resp.status;
    let body: TokenResponse = resp
        .json()
        .map_err(|_| unexpected("Microsoft", format!("HTTP {status} refreshing the sign-in")))?;

    if let Some(access) = body.access_token {
        return Ok(MsTokens {
            access_token: SecretString::new(access),
            refresh_token: body.refresh_token.map(SecretString::new),
        });
    }
    match body.error.as_deref() {
        Some("invalid_grant" | "interaction_required" | "consent_required") => {
            Err(AuthError::ReauthRequired.into())
        }
        other => Err(unexpected(
            "Microsoft",
            format!("error {}", short(other.unwrap_or("none"))),
        )),
    }
}

use zeroize::{Zeroize, ZeroizeOnDrop};

use crate::redact;

/// A token that can't be printed by accident: `Debug` hides it, it is wiped from memory
/// when dropped (best effort: copies made by HTTP buffers are outside our control), and it
/// is registered with the log redactor the moment it exists.
#[derive(Clone, Zeroize, ZeroizeOnDrop)]
pub struct SecretString(String);

impl SecretString {
    pub fn new(value: impl Into<String>) -> Self {
        let value = value.into();
        redact::global().register_secret(&value);
        Self(value)
    }

    /// The raw value. Use only to build a request or a launch argument.
    pub fn expose(&self) -> &str {
        &self.0
    }
}

impl std::fmt::Debug for SecretString {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.write_str("SecretString([REDACTED])")
    }
}

impl AuthService {
    pub fn new(db: Db, keys: Arc<dyn SecretStore>, client_id: Option<String>) -> Result<Self> {
        let http = HttpClient::new(&format!("{APP_NAME}/{APP_VERSION}"), host_policy())?;
        Ok(Self::build(
            db,
            http,
            keys,
            Endpoints::default(),
            client_id,
            Timing::default(),
        ))
    }

    fn build(
        db: Db,
        http: HttpClient,
        keys: Arc<dyn SecretStore>,
        endpoints: Endpoints,
        client_id: Option<String>,
        timing: Timing,
    ) -> Self {
        Self {
            db,
            http,
            keys,
            endpoints,
            client_id: client_id.filter(|id| valid_client_id(id)),
            timing,
            flows: Mutex::new(Flows::default()),
            sessions: Mutex::new(HashMap::new()),
            refresh_lock: tokio::sync::Mutex::new(()),
        }
    }

    /// Register known usernames with the log redactor. Call once at startup.
    pub async fn init(&self) -> Result<()> {
        for account in accounts::list(&self.db).await? {
            redact::global().register_secret(&account.username);
        }
        Ok(())
    }

    fn client_id(&self) -> Result<&str> {
        self.client_id
            .as_deref()
            .ok_or_else(|| AuthError::NotConfigured.into())
    }

    // ---- keychain access (blocking OS calls run off the async runtime) ----

    async fn keys_get(&self, id: &str) -> Result<Option<Vec<u8>>> {
        let (keys, id) = (Arc::clone(&self.keys), id.to_owned());
        tokio::task::spawn_blocking(move || keys.get(&id))
            .await
            .map_err(|e| Error::Keychain(e.to_string()))?
    }

    async fn keys_set(&self, id: &str, secret: &[u8]) -> Result<()> {
        let (keys, id, secret) = (Arc::clone(&self.keys), id.to_owned(), secret.to_vec());
        tokio::task::spawn_blocking(move || keys.set(&id, &secret))
            .await
            .map_err(|e| Error::Keychain(e.to_string()))?
    }

    async fn keys_delete(&self, id: &str) -> Result<()> {
        let (keys, id) = (Arc::clone(&self.keys), id.to_owned());
        tokio::task::spawn_blocking(move || keys.delete(&id))
            .await
            .map_err(|e| Error::Keychain(e.to_string()))?
    }

    // ---- sign-in (device code flow) ----

    /// Ask Microsoft for a code the user enters on their own device or browser.
    pub async fn begin_device_flow(&self) -> Result<DeviceCodeInfo> {
        let client_id = self.client_id()?;
        let code = ms::request_device_code(&self.http, &self.endpoints, client_id).await?;
        let info = DeviceCodeInfo {
            user_code: code.user_code,
            verification_uri: code.verification_uri.clone(),
            expires_in: code.expires_in.as_secs(),
        };

        let mut flows = lock(&self.flows);
        if let Some(old) = flows.current.take() {
            old.cancel.notify_one();
        }
        flows.next_id += 1;
        let id = flows.next_id;
        flows.current = Some(PendingFlow {
            id,
            device_code: code.device_code,
            verification_uri: code.verification_uri,
            expires_at: Instant::now() + code.expires_in,
            interval: code.interval,
            cancel: Arc::new(Notify::new()),
        });
        Ok(info)
    }

    /// The (validated) page where the user enters the code, for the "open in browser" button.
    pub fn pending_verification_uri(&self) -> Option<String> {
        lock(&self.flows)
            .current
            .as_ref()
            .map(|f| f.verification_uri.clone())
    }

    /// Wait for the user to finish signing in, then complete the Xbox and Minecraft steps.
    /// Returns when the flow succeeds, is declined, expires, or is cancelled.
    pub async fn poll(&self) -> Result<Account> {
        let (flow_id, device_code, expires_at, interval, cancel) = {
            let flows = lock(&self.flows);
            let f = flows.current.as_ref().ok_or(AuthError::NoPendingFlow)?;
            (
                f.id,
                f.device_code.clone(),
                f.expires_at,
                f.interval,
                Arc::clone(&f.cancel),
            )
        };
        let result = self
            .poll_loop(&device_code, expires_at, interval, &cancel)
            .await;

        let mut flows = lock(&self.flows);
        if flows.current.as_ref().is_some_and(|f| f.id == flow_id) {
            flows.current = None;
        }
        result
    }

    async fn poll_loop(
        &self,
        device_code: &SecretString,
        expires_at: Instant,
        interval: Duration,
        cancel: &Notify,
    ) -> Result<Account> {
        let client_id = self.client_id()?;
        let mut extra = Duration::ZERO;
        loop {
            let wait = self.timing.interval_override.unwrap_or(interval) + extra;
            tokio::select! {
                () = cancel.notified() => return Err(AuthError::FlowCancelled.into()),
                () = tokio::time::sleep(wait) => {}
            }
            if Instant::now() >= expires_at {
                return Err(AuthError::FlowExpired.into());
            }
            match ms::poll_token(&self.http, &self.endpoints, client_id, device_code).await? {
                PollOutcome::Tokens(tokens) => return self.finish_sign_in(tokens).await,
                PollOutcome::Pending => {}
                PollOutcome::SlowDown => extra += self.timing.slow_down_step,
            }
        }
    }

    /// Abandon the sign-in in progress, if any.
    pub fn cancel_flow(&self) {
        if let Some(flow) = lock(&self.flows).current.take() {
            flow.cancel.notify_one();
        }
    }

    async fn minecraft_login(&self, ms_access: &SecretString) -> Result<(McLogin, Profile)> {
        let xbl = xbox::authenticate(&self.http, &self.endpoints, ms_access).await?;
        let xsts = xbox::authorize_xsts(&self.http, &self.endpoints, &xbl).await?;
        let login = minecraft::login_with_xbox(&self.http, &self.endpoints, &xsts).await?;
        let profile =
            minecraft::fetch_profile(&self.http, &self.endpoints, &login.access_token).await?;
        Ok((login, profile))
    }

    fn cache_session(&self, profile: &Profile, login: &McLogin) {
        redact::global().register_secret(&profile.name);
        lock(&self.sessions).insert(
            profile.id.clone(),
            CachedSession {
                access_token: login.access_token.clone(),
                username: profile.name.clone(),
                expires_at: Instant::now() + login.expires_in,
            },
        );
    }

    async fn finish_sign_in(&self, tokens: MsTokens) -> Result<Account> {
        let refresh = tokens
            .refresh_token
            .ok_or_else(|| unexpected("Microsoft", "no refresh token was issued"))?;
        // Ownership is checked here: a missing profile aborts before anything is stored.
        let (login, profile) = self.minecraft_login(&tokens.access_token).await?;

        // Keychain first: if it fails, no account row is created.
        self.keys_set(&profile.id, refresh.expose().as_bytes())
            .await?;
        let account = accounts::upsert_signed_in(&self.db, &profile, now_unix()).await?;
        self.cache_session(&profile, &login);
        tracing::info!("signed in to a Minecraft account");
        Ok(account)
    }

    // ---- sessions for launching ----

    fn cached(&self, id: &str) -> Option<LaunchSession> {
        let sessions = lock(&self.sessions);
        let s = sessions.get(id)?;
        (s.expires_at > Instant::now() + REFRESH_MARGIN).then(|| LaunchSession {
            account_id: id.to_owned(),
            username: s.username.clone(),
            access_token: s.access_token.clone(),
        })
    }

    /// A valid Minecraft session for this account, refreshing through Microsoft if the
    /// cached one is missing or about to expire. Only the launch pipeline calls this.
    pub async fn session_for_launch(&self, account_id: &str) -> Result<LaunchSession> {
        validate_account_id(account_id)?;
        if let Some(session) = self.cached(account_id) {
            return Ok(session);
        }
        let _guard = self.refresh_lock.lock().await;
        // Another task may have refreshed while we waited.
        if let Some(session) = self.cached(account_id) {
            return Ok(session);
        }
        self.refresh_account(account_id).await
    }

    async fn refresh_account(&self, id: &str) -> Result<LaunchSession> {
        let client_id = self.client_id()?;
        if accounts::get(&self.db, id).await?.is_none() {
            return Err(AuthError::AccountNotFound.into());
        }
        let stored = self
            .keys_get(id)
            .await?
            .and_then(|raw| String::from_utf8(raw).ok());
        let Some(stored) = stored else {
            accounts::mark_needs_reauth(&self.db, id).await?;
            return Err(AuthError::ReauthRequired.into());
        };

        let tokens = match ms::refresh(
            &self.http,
            &self.endpoints,
            client_id,
            &SecretString::new(stored),
        )
        .await
        {
            Ok(tokens) => tokens,
            Err(e @ Error::Auth(AuthError::ReauthRequired)) => {
                accounts::mark_needs_reauth(&self.db, id).await?;
                return Err(e);
            }
            Err(e) => return Err(e),
        };
        if let Some(rotated) = &tokens.refresh_token {
            self.keys_set(id, rotated.expose().as_bytes()).await?;
        }

        let (login, profile) = self.minecraft_login(&tokens.access_token).await?;
        if profile.id != id {
            return Err(unexpected(
                "Minecraft",
                "the refreshed profile belongs to a different account",
            ));
        }
        accounts::update_profile(&self.db, &profile).await?;
        self.cache_session(&profile, &login);
        Ok(LaunchSession {
            account_id: id.to_owned(),
            username: profile.name,
            access_token: login.access_token,
        })
    }

    // ---- account management ----

    pub async fn list_accounts(&self) -> Result<Vec<Account>> {
        accounts::list(&self.db).await
    }

    pub async fn set_active(&self, id: &str) -> Result<Account> {
        validate_account_id(id)?;
        accounts::set_active(&self.db, id, now_unix()).await
    }

    /// Remove an account and its stored sign-in. The keychain entry goes first: if that
    /// fails the account stays listed so the user can retry, instead of leaving an
    /// orphaned secret behind.
    pub async fn remove_account(&self, id: &str) -> Result<()> {
        validate_account_id(id)?;
        if accounts::get(&self.db, id).await?.is_none() {
            return Err(AuthError::AccountNotFound.into());
        }
        self.keys_delete(id).await?;
        accounts::remove(&self.db, id).await?;
        lock(&self.sessions).remove(id);
        Ok(())
    }
}

    #[derive(Clone, Default)]
    struct LogBuf(Arc<Mutex<Vec<u8>>>);

    impl Write for LogBuf {
        fn write(&mut self, data: &[u8]) -> std::io::Result<usize> {
            self.0.lock().unwrap().extend_from_slice(data);
            Ok(data.len())
        }
        fn flush(&mut self) -> std::io::Result<()> {
            Ok(())
        }
    }

    impl<'a> MakeWriter<'a> for LogBuf {
        type Writer = LogBuf;
        fn make_writer(&'a self) -> LogBuf {
            self.clone()
        }
    }

    /// Capture every event from every crate at TRACE, with NO redaction, so tests prove the
    /// code never emits a secret rather than relying on the redactor to hide it.
    fn capture_logs() -> (tracing::subscriber::DefaultGuard, LogBuf) {
        let buf = LogBuf::default();
        let subscriber = tracing_subscriber::fmt()
            .with_ansi(false)
            .with_max_level(tracing::Level::TRACE)
            .with_writer(buf.clone())
            .finish();
        (tracing::subscriber::set_default(subscriber), buf)
    }

    fn contains(haystack: &[u8], needle: &[u8]) -> bool {
        haystack.windows(needle.len()).any(|w| w == needle)
    }

//! Xbox Live user token and XSTS authorization for the Minecraft relying party.

use reqwest::Method;
use serde::Deserialize;
use serde_json::json;

use super::endpoints::Endpoints;
use super::error::{unexpected, AuthError};
use super::secret::SecretString;
use crate::net::{Body, HttpClient};
use crate::Result;

pub struct XboxToken {
    pub token: SecretString,
    /// "User hash": identifies the Xbox user inside the Minecraft identity token.
    pub uhs: String,
}

#[derive(Deserialize)]
struct XboxResponse {
    #[serde(rename = "Token")]
    token: String,
    #[serde(rename = "DisplayClaims")]
    claims: Claims,
}

#[derive(Deserialize)]
struct Claims {
    xui: Vec<Xui>,
}

#[derive(Deserialize)]
struct Xui {
    uhs: String,
}

#[derive(Deserialize)]
struct XstsError {
    #[serde(rename = "XErr")]
    code: Option<u64>,
}

fn parse(resp: &crate::net::HttpResponse, what: &str) -> Result<XboxToken> {
    let body: XboxResponse = resp.json()?;
    let uhs = body
        .claims
        .xui
        .into_iter()
        .next()
        .map(|x| x.uhs)
        .filter(|u| (1..=64).contains(&u.len()) && u.bytes().all(|b| b.is_ascii_alphanumeric()))
        .ok_or_else(|| unexpected("Xbox", format!("{what} response had no valid user hash")))?;
    Ok(XboxToken {
        token: SecretString::new(body.token),
        uhs,
    })
}

pub async fn authenticate(
    http: &HttpClient,
    ep: &Endpoints,
    ms_access: &SecretString,
) -> Result<XboxToken> {
    let body = json!({
        "Properties": {
            "AuthMethod": "RPS",
            "SiteName": "user.auth.xboxlive.com",
            "RpsTicket": format!("d={}", ms_access.expose()),
        },
        "RelyingParty": "http://auth.xboxlive.com",
        "TokenType": "JWT",
    });
    let resp = http
        .execute(Method::POST, &ep.xbox_user_auth, Body::Json(&body), None)
        .await?;
    if !resp.is_success() {
        return Err(unexpected(
            "Xbox",
            format!("HTTP {} from user authentication", resp.status),
        ));
    }
    parse(&resp, "user authentication")
}

pub async fn authorize_xsts(
    http: &HttpClient,
    ep: &Endpoints,
    user: &XboxToken,
) -> Result<XboxToken> {
    let body = json!({
        "Properties": {
            "SandboxId": "RETAIL",
            "UserTokens": [user.token.expose()],
        },
        "RelyingParty": "rp://api.minecraftservices.com/",
        "TokenType": "JWT",
    });
    let resp = http
        .execute(Method::POST, &ep.xsts, Body::Json(&body), None)
        .await?;
    if resp.status == 401 {
        let code = resp.json::<XstsError>().ok().and_then(|e| e.code);
        return Err(map_xsts_error(code).into());
    }
    if !resp.is_success() {
        return Err(unexpected(
            "Xbox",
            format!("HTTP {} from XSTS", resp.status),
        ));
    }
    parse(&resp, "XSTS")
}

/// <https://wiki.vg/Microsoft_Authentication_Scheme> documents these `XErr` codes.
fn map_xsts_error(code: Option<u64>) -> AuthError {
    match code {
        Some(2_148_916_233) => AuthError::XboxNoAccount,
        Some(2_148_916_235) => AuthError::XboxRegionBlocked,
        Some(2_148_916_236 | 2_148_916_237) => AuthError::XboxAdultVerification,
        Some(2_148_916_238) => AuthError::XboxChildAccount,
        Some(other) => AuthError::XboxOther(other),
        None => AuthError::XboxOther(0),
    }
}

//! The only HTTP client in Kiln. It speaks HTTPS only, refuses any host that isn't on an
//! explicit allowlist, never follows redirects on its own, caps response sizes, and never
//! logs URLs beyond host and path or any request or response body.

use std::sync::Once;
use std::time::Duration;

use reqwest::{Method, Url};
use serde::de::DeserializeOwned;

use crate::{Error, Result};

/// Largest response body accepted by [`HttpClient::execute`].
pub const MAX_RESPONSE_BYTES: usize = 2 * 1024 * 1024;

/// Which hosts a client may talk to. Entries are exact host names, or start with `.` to
/// also allow subdomains (`.example.com` matches `example.com` and `a.example.com`, but
/// not `badexample.com` or `example.com.evil.net`).
#[derive(Debug, Clone)]
pub struct HostPolicy {
    allowed: Vec<String>,
    #[cfg(test)]
    loopback_http: bool,
}

fn host_matches(host: &str, entry: &str) -> bool {
    match entry.strip_prefix('.') {
        Some(bare) => host == bare || host.ends_with(entry),
        None => host == entry,
    }
}

impl HostPolicy {
    pub fn new(allowed: &[&str]) -> Self {
        Self {
            allowed: allowed.iter().map(|h| h.to_ascii_lowercase()).collect(),
            #[cfg(test)]
            loopback_http: false,
        }
    }

    /// Test-only: also allow plain HTTP to a loopback mock server.
    #[cfg(test)]
    pub fn with_loopback_http(mut self) -> Self {
        self.loopback_http = true;
        self
    }

    fn loopback_http_enabled(&self) -> bool {
        #[cfg(test)]
        {
            self.loopback_http
        }
        #[cfg(not(test))]
        {
            false
        }
    }

    pub fn check(&self, url: &Url) -> Result<()> {
        let deny =
            |why: &str| Error::HostNotAllowed(format!("{} ({why})", url.host_str().unwrap_or("?")));
        let host = url.host_str().ok_or_else(|| deny("no host"))?;

        if self.loopback_http_enabled()
            && url.scheme() == "http"
            && matches!(host, "127.0.0.1" | "localhost")
        {
            return Ok(());
        }
        if url.scheme() != "https" {
            return Err(deny("not HTTPS"));
        }
        if !url.username().is_empty() || url.password().is_some() {
            return Err(deny("credentials in URL"));
        }
        if url.port().is_some_and(|p| p != 443) {
            return Err(deny("unexpected port"));
        }
        if self.allowed.iter().any(|entry| host_matches(host, entry)) {
            Ok(())
        } else {
            Err(deny("host not on the allowlist"))
        }
    }
}

pub enum Body<'a> {
    Empty,
    Form(&'a [(&'a str, &'a str)]),
    Json(&'a serde_json::Value),
}

/// Status and (size-capped) body. `Debug` deliberately hides the body: it may hold tokens.
pub struct HttpResponse {
    pub status: u16,
    body: Vec<u8>,
}

impl std::fmt::Debug for HttpResponse {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        write!(
            f,
            "HttpResponse {{ status: {}, body: {} bytes }}",
            self.status,
            self.body.len()
        )
    }
}

impl HttpResponse {
    pub fn is_success(&self) -> bool {
        (200..300).contains(&self.status)
    }

    pub fn json<T: DeserializeOwned>(&self) -> Result<T> {
        serde_json::from_slice(&self.body)
            .map_err(|_| Error::Http("the server sent a response in an unexpected format".into()))
    }
}

fn http_err(e: reqwest::Error) -> Error {
    // Strip the URL: it could carry query parameters.
    Error::Http(e.without_url().to_string())
}

fn install_crypto_provider() {
    static ONCE: Once = Once::new();
    ONCE.call_once(|| {
        // Err means another provider is already installed, which is fine.
        let _ = rustls::crypto::ring::default_provider().install_default();
    });
}

#[derive(Debug, Clone)]
pub struct HttpClient {
    inner: reqwest::Client,
    policy: HostPolicy,
}

impl HttpClient {
    pub fn new(user_agent: &str, policy: HostPolicy) -> Result<Self> {
        install_crypto_provider();
        let mut builder = reqwest::Client::builder()
            .user_agent(user_agent)
            .https_only(!policy.loopback_http_enabled())
            .redirect(reqwest::redirect::Policy::none())
            .connect_timeout(Duration::from_secs(10))
            .timeout(Duration::from_secs(30));
        if policy.loopback_http_enabled() {
            // A system proxy must never intercept calls to a local mock server.
            builder = builder.no_proxy();
        }
        Ok(Self {
            inner: builder.build().map_err(http_err)?,
            policy,
        })
    }

    pub async fn execute(
        &self,
        method: Method,
        url: &str,
        body: Body<'_>,
        bearer: Option<&str>,
    ) -> Result<HttpResponse> {
        let url = Url::parse(url).map_err(|_| Error::HostNotAllowed("invalid URL".into()))?;
        self.policy.check(&url)?;

        let mut request = self.inner.request(method.clone(), url.clone());
        request = match body {
            Body::Empty => request,
            Body::Form(fields) => request.form(fields),
            Body::Json(value) => request.json(value),
        };
        if let Some(token) = bearer {
            request = request.bearer_auth(token);
        }

        let mut response = request.send().await.map_err(http_err)?;
        let status = response.status().as_u16();
        tracing::debug!(
            %method,
            host = url.host_str().unwrap_or(""),
            path = url.path(),
            status,
            "http response"
        );

        let mut body = Vec::new();
        while let Some(chunk) = response.chunk().await.map_err(http_err)? {
            if body.len() + chunk.len() > MAX_RESPONSE_BYTES {
                return Err(Error::Http("the server's response was too large".into()));
            }
            body.extend_from_slice(&chunk);
        }
        Ok(HttpResponse { status, body })
    }
}
